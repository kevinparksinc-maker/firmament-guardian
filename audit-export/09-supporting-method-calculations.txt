/** Degree aspect strength, BPHS 26.6–12. No Western square/trine polarity is used. */
export function sphutaDrishti(source: Graha, from: number, to: number): number {
  const d = norm(to - from),
    r = d % 30,
    band = Math.floor(d / 30);
  if (source === "Saturn") {
    if (band === 1) return 2 * r;
    if (band === 2) return 60 - r / 2;
    if (band === 8) return 30 + r;
    if (band === 9) return 60 - 2 * r;
  }
  if (source === "Mars") {
    if (band === 2) return 15 + 1.5 * r;
    if (band === 3 || band === 7) return 60 - r;
    if (band === 6) return 60;
  }
  if (source === "Jupiter") {
    if (band === 3 || band === 7) return 45 + r / 2;
    if (band === 4 || band === 8) return 60 - 2 * r;
  }
  if (d < 30 || d >= 300) return 0;
  if (d < 60) return (d - 30) / 2;
  if (d < 90) return d - 45;
  if (d < 120) return (120 - d) / 2 + 30;
  if (d < 150) return 150 - d;
  if (d < 180) return 2 * (d - 150);
  return (300 - d) / 2;
}

/** The three missing Saptavarga projections are local derivatives, not new skies. */
export function strengthVargaSign(longitude: number, division: number): number {
  const s = sign(longitude),
    degree = norm(longitude) % 30;
  if (division === 1) return s;
  if (division === 2 || division === 3 || division === 9)
    return vargaPlacement(longitude, `D${division}` as "D2" | "D3" | "D9")
      .signIndex;
  if (division === 7)
    return (s + (s % 2 === 0 ? 0 : 6) + Math.floor(degree / (30 / 7))) % 12;
  if (division === 12) return (s + Math.floor(degree / 2.5)) % 12;
  if (division === 30) {
    const bounds = s % 2 === 0 ? [5, 10, 18, 25, 30] : [5, 12, 20, 25, 30];
    const signs = s % 2 === 0 ? [0, 10, 8, 2, 6] : [1, 5, 11, 9, 7];
    return signs[bounds.findIndex(end => degree < end)];
  }
  throw new Error(`Unsupported strength division D${division}`);
}
export function saptavargajaBala(
  body: Graha,
  positions: Record<Graha, number>

    const s = strengthVargaSign(positions[body], division),
      lord = LORDS[s];

      value: number;
    if (s === MOOLA[body]) {
      value = 45;
      relation = "moolatrikona";

      relation = "own sign";

          ? -1

      const temporary = [2, 3, 4, 10, 11, 12].includes(offset) ? 1 : -1;
      const compound = natural + temporary;

          string,

export function ayanaBala(body: Graha, declination: number) {
  // Classical declination approximation has a 24° normalizing amplitude.
  const signed =
    body === "Mercury"
      ? Math.abs(declination)
      : ["Moon", "Saturn"].includes(body)
        ? -declination

function temporal(chart: ChartResult) {
  const t = Date.parse(chart.utc),
    latitude = chart.input.latitude,
    longitude = chart.input.longitude;
  assert(
    Number.isFinite(t) &&
      Number.isFinite(latitude) &&
      Number.isFinite(longitude),
    "Resolved instant and observer coordinates required."
  );
  const base = Date.UTC(
    new Date(t).getUTCFullYear(),
    new Date(t).getUTCMonth(),
    new Date(t).getUTCDate()
  );
  const candidates = [-2, -1, 0, 1, 2]
    .map(offset => {
      const date = new Date(base + offset * 86400000)
        .toISOString()
        .slice(0, 10);
      return {
        rise: noaaSunriseUtc(date, latitude, longitude).getTime(),
        set: noaaSunriseUtc(date, latitude, longitude, true).getTime(),
      };
    })
    .sort((a, b) => a.rise - b.rise);
  const index = candidates.findIndex(
    (v, i) => v.rise <= t && candidates[i + 1]?.rise > t
  );
  assert(index >= 0, "Sunrise interval unavailable.");
  const current = candidates[index],
    next = candidates[index + 1];
  const day = t < current.set,
    phase = day
      ? (t - current.rise) / (current.set - current.rise)
      : (t - current.set) / (next.rise - current.set);
  const third = Math.min(2, Math.floor(phase * 3));
  const thirdLord: Graha = (
    day ? ["Mercury", "Sun", "Saturn"] : ["Moon", "Venus", "Mars"]
  )[third] as Graha;
  const dateParts = Object.fromEntries(
    new Intl.DateTimeFormat("en-CA", {
      timeZone: chart.input.timezone,
      year: "numeric",
      month: "numeric",
      day: "numeric",
    })
      .formatToParts(new Date(current.rise))
      .map(p => [p.type, p.value])
  );
  const year = Number(dateParts.year),
    month = Number(dateParts.month),
    date = Number(dateParts.day);
  const weekday = new Date(Date.UTC(year, month - 1, date)).getUTCDay(),
    dayLord = WEEK[weekday];
  const horaIndex = (day ? 0 : 12) + Math.min(11, Math.floor(phase * 12));
  const horaLord = HORA[(HORA.indexOf(dayLord) + horaIndex) % 7];
  const noon = (current.rise + current.set) / 2;
  const solarHours = mod(12 + (t - noon) / 3600000, 24);
  return {
    sunriseUtc: new Date(current.rise).toISOString(),
    sunsetUtc: new Date(current.set).toISOString(),
    nextSunriseUtc: new Date(next.rise).toISOString(),
    day,
    third: third + 1,
    thirdLord,
    dayLord,
    horaLord,
    hora: horaIndex + 1,
    yearLord: WEEK[mod((year + 3) * 3, 7)],
    monthLord: WEEK[mod((12 * year + month + 6) * 2, 7)],
    solarHours,
    twilight:
      Math.min(
        Math.abs(t - current.rise),
        Math.abs(t - current.set),
        Math.abs(t - next.rise)
      ) <=
      24 * 60000,
  };
}

/** Pure continuous Cheshta formula. Use a circular midpoint across the 0° seam. */
export function cheshtaKendra(
  body: Graha,
  trueLongitude: number,
  meanLongitude: number,
  sunMean: number
) {
  const inner = body === "Mercury" || body === "Venus";
  const mean = inner ? sunMean : meanLongitude,
    apex = inner ? meanLongitude : sunMean;
  const midpoint = norm(mean + (norm(trueLongitude - mean + 180) - 180) / 2);
  return angle(apex, midpoint) / 3;
}
function meanOrbitalLongitude(jd: number, id: number) {
  // M and true anomaly are scalar angles in the orbital plane; none of the
  // orbital-element ecliptic longitudes are used as chart coordinates.
  const orbit = sweph.get_orbital_elements(jd + sweph.deltat(jd), id, 4);
  assert(
    orbit.flag >= 0 && orbit.data?.slice(6, 8).every(Number.isFinite),
    `Orbital anomalies missing for ${id}.`
  );
  const { xyz, velocity } = ephemerisXYZ(jd, id, 8 | 16);
  const r = [xyz.x, xyz.y, xyz.z],
    v = velocity;
  const h = [
    r[1] * v[2] - r[2] * v[1],
    r[2] * v[0] - r[0] * v[2],
    r[0] * v[1] - r[1] * v[0],
  ];
  const length = Math.hypot(...h);
  assert(length > 0, "Orbital normal cannot be zero.");
  const n = h.map(x => x / length),
    d = (orbit.data[6] - orbit.data[7]) * rad;
  const cross = [
      n[1] * r[2] - n[2] * r[1],
      n[2] * r[0] - n[0] * r[2],
      n[0] * r[1] - n[1] * r[0],
    ],
    dot = sum(n.map((x, i) => x * r[i]));
  const rotated = r.map(
    (x, i) =>
      x * Math.cos(d) + cross[i] * Math.sin(d) + n[i] * dot * (1 - Math.cos(d))
  );
  return norm(Math.atan2(rotated[1], rotated[0]) / rad);
}

/**
 * VEDIC CHART READER — one authoritative saved natal chart, many derived charts.
 *
 * This service does not accept a birth date, time, coordinates or an ephemeris.
 * It does not calculate sunrise, planetary positions or the natal Ascendant.
 * The profile repository must supply its already-calculated Master Natal
 * Snapshot. The five charts are mathematical projections of THAT snapshot.
 *
 * D1: preserve natal longitude, Ascendant and 12 equal-house cusps.
 * D2: divide signs into 2 × 15° Hora sections (Sun/Leo, Moon/Cancer).
 * D3: divide signs into 3 × 10° Drekkana sections (1st/5th/9th).
 * D9: divide signs into 9 × 3°20′ Navamsha sections (modality starts).
 * D10: divide signs into 10 × 3° Dashamsha sections (odd/even starts).
 *
 * All D2-D10 planets and divisional Ascendants are transformed from natal
 * longitude; divisional houses are projected from the transformed Ascendant.
 * This is the application's explicit fixed-frame variant, not an implicit
 * Lahiri conversion. A D-chart is NOT a second astronomical birth chart.
 *
 * Display consumers receive chartLayout = north-indian-diamond, house cusps,
 * numbered houses, Ascendant, placements, house assignments and formula
 * traces. Interpretations must consume these computed results, not guess.
 */
export function vedicChartsFromMasterNatal(snapshot: UnifiedMasterChartSnapshot) {
  if (snapshot.inputs.mode !== "natal") {
    throw new Error("Vedic divisional charts require the authoritative natal snapshot, not a transit chart.");
  }
  const state = snapshot.baseCelestialState;
  if (!state?.ascendant || !Array.isArray(state.planets) || state.houses?.length !== 12) {
    throw new Error("Master natal chart is missing its Ascendant, planets or 12 houses.");
  }
  const charts = buildDivisionalCharts({
    ascendantLongitude: state.ascendant.longitude,
    planets: [
      ...state.planets.map(p => ({ name: p.name, longitude: p.longitude })),
      { name: state.northNode.name, longitude: state.northNode.longitude },
      { name: state.southNode.name, longitude: state.southNode.longitude },
    ],
    d1HouseCusps: state.houses.map(h => h.cusp),
  });
  return {
    sourceSnapshotId: snapshot.snapshotId,
    source: "profile-master-natal" as const,
    astronomicalRecalculation: false as const,
    charts,
    byDivision: Object.fromEntries(charts.map(chart => [chart.division, chart])) as Record<Division, (typeof charts)[number]>,
  };
}

/**
 * The caller owns the profile persistence/cache. Fetch the existing snapshot;
 * do not call getMasterNatalChart() or calculateChart() from this service.
 */
export async function loadProfileVedicCharts(
  profileId: string,
  loadSavedNatalSnapshot: (profileId: string) => Promise<UnifiedMasterChartSnapshot | null>,
) {
  const snapshot = await loadSavedNatalSnapshot(profileId);
  if (!snapshot) throw new Error("Saved profile natal chart not found. Generate and save D1 before requesting Vargas.");
  return vedicChartsFromMasterNatal(snapshot);
}

    ) ||
    ["Sun", "Moon", "Ascendant", "Saturn", "Jupiter", "Venus"].includes(
      row.name
    )
  );
}

        sourceTitle: "Firmament architecture",
        retrievedAt: new Date().toISOString(),
        ruleId: "western-5-degree-geometry",
        confidence: "high",
      },
    });
  }
  return evidence;
}

import { normalizeLongitude } from "../../shared/hybrid";
import { ArabicSnapshotSchema, type ArabicSnapshot } from "../patterns/types";

export type Sect = "day" | "night";
export type LotFormula = { name: string; add: string; subtract: string };

export type LotInputs = {
  ascendant: number;
  sun: number;
  moon: number;
  venus?: number;
  mercury?: number;
  mars?: number;
  jupiter?: number;
  sect: Sect;
};

export const LOT_FORMULAS: Record<string, LotFormula> = {
  fortune: { name: "Lot of Fortune", add: "moon", subtract: "sun" },
  spirit: { name: "Lot of Spirit", add: "sun", subtract: "moon" },
  eros: { name: "Lot of Eros", add: "venus", subtract: "spirit" },
  necessity: { name: "Lot of Necessity", add: "fortune", subtract: "mercury" },
};

function requireValue(inputs: Record<string, number | undefined>, key: string) {
  const value = inputs[key];
  if (value == null || !Number.isFinite(value))
    throw new Error(`Arabic Lot calculation requires ${key}.`);
  return normalizeLongitude(value);
}

/**
 * Calculates a traditional Lot from Ascendant + addend - subtractend.
 * Fortune and Spirit swap their luminary order by sect; callers may pass an
 * explicit formula for other Lots so the convention remains visible and testable.
 */
export function calculateLot(inputs: LotInputs, formula: LotFormula): number {
  const values: Record<string, number | undefined> = {
    ascendant: inputs.ascendant,
    sun: inputs.sun,
    moon: inputs.moon,
    venus: inputs.venus,
    mercury: inputs.mercury,
    mars: inputs.mars,
    jupiter: inputs.jupiter,
  };
  if (formula.add === "spirit")
    values.spirit = calculateSectLot(inputs, LOT_FORMULAS.spirit);
  if (formula.add === "fortune")
    values.fortune = calculateSectLot(inputs, LOT_FORMULAS.fortune);
  if (formula.subtract === "spirit")
    values.spirit = values.spirit ?? calculateSectLot(inputs, LOT_FORMULAS.spirit);
  if (formula.subtract === "fortune")
    values.fortune =
      values.fortune ?? calculateSectLot(inputs, LOT_FORMULAS.fortune);
  return normalizeLongitude(
    requireValue(values, "ascendant") +
      requireValue(values, formula.add) -
      requireValue(values, formula.subtract)
  );
}

function calculateSectLot(inputs: LotInputs, formula: LotFormula): number {
  const { add, subtract } = formula;
  return calculateLot(inputs, inputs.sect === "day" ? formula : { ...formula, add: subtract, subtract: add });
}

export function calculateArabicLots(inputs: LotInputs): Record<string, number> {
  const results: Record<string, number> = {
    fortune: calculateSectLot(inputs, LOT_FORMULAS.fortune),
    spirit: calculateSectLot(inputs, LOT_FORMULAS.spirit),
  };
  if (Number.isFinite(inputs.venus))
    results.eros = calculateSectLot(inputs, LOT_FORMULAS.eros);
  if (Number.isFinite(inputs.mercury))
    results.necessity = calculateSectLot(inputs, LOT_FORMULAS.necessity);
  return Object.fromEntries(
    Object.entries(results).map(([name, longitude]) => [
      name,
      Number(longitude.toFixed(6)),
    ])
  );
}

export interface ArabicProvider {
  readonly name: string;
  getAnalysis(input: LotInputs): Promise<ArabicSnapshot>;
}

export class MockArabicProvider implements ArabicProvider {
  readonly name = "mock-canopy";
  async getAnalysis(input: LotInputs): Promise<ArabicSnapshot> {
    return ArabicSnapshotSchema.parse({
      provider: this.name,
      lots: calculateArabicLots(input),
      signals: [],
    });
  }
}

export type Fact = { id: string; layerId: string; value: unknown };

/** Select planet-specific findings once; never ship the 575k-character life-story context. */
export function buildPlacementPackets(
  chart: ChartResult,
  synthesis: SynthesisResult,
  options: { requireJudgments?: boolean } = {}
) {
  const sun = chart.movingBodies.find(p => p.name === "Sun");
  const personaMatrix = buildPlanetPersonaMatrix(chart);
  return placementRows(chart).map(row => {
    const dossier = retrievePlacementDossier(
      row,
      chart,
      Boolean(sun && sun.house >= 7)
    );
    const escaped = row.name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    const bodyMatch = new RegExp(`(^|[^A-Za-z])${escaped}([^A-Za-z]|$)`, "i");
    const slug = row.name.toLowerCase().replace(/\s+/g, "-");
    const facts: Fact[] = [
      {
        id: `planet-${slug}`,
        layerId: "western-planets-and-signs",
        value: {
          display: row.display,
          longitude: row.longitude,
          house: row.house,
          godHouse: row.godHouse,
          agentHouse: row.agentHouse,
          retrograde: row.retrograde,
        },
      },
    ];
    const judgments = synthesis.methodOutcomes
      .filter(
        o => o.status === "interpreted" && !o.methodId.startsWith("transit:")
      )
      .flatMap(outcome =>
        outcome.findingJudgments
          .filter(
            f =>
              (!outcome.methodId.includes("dasha-and-ashtakavarga") ||
                f.subject.startsWith("SAV ")) &&
              (bodyMatch.test(f.subject) ||
                f.evidenceIds.some(
                  id => id.includes(`-${slug}-`) || id.endsWith(`-${slug}`)
                ) ||
                f.subject === `House ${row.agentHouse ?? row.house}` ||
                f.subject === `SAV ${row.display.split(" ")[0]}`)
          )
          .map(f => ({
            findingId: f.id,
            methodId: outcome.methodId,
            frame: f.frame ?? outcome.frame,
            judgment: f.judgment,
            mechanism: f.mechanism,
            evidenceIds: f.evidenceIds,
            sources: Array.from(new Set(f.rules.map(r => r.sourceTitle))),
            limitation: f.limitation,
          }))
      );
    if (!judgments.length && options.requireJudgments !== false)
      throw new Error(`PLACEMENT_JUDGMENTS_MISSING: ${row.name}`);
    for (const j of judgments)
      for (const id of j.evidenceIds) {
        if (!facts.some(f => f.id === id && f.layerId === j.methodId))
          facts.push({
            id,
            layerId: j.methodId,
            value: {
              judgment: j.judgment,
              mechanism: j.mechanism,
              frame: j.frame,
              sources: j.sources,
              limitation: j.limitation,
            },
          });
      }
    const persona = personaMatrix.find(p => p.planet === row.name);
    return {
      name: row.name,
      placementEvidenceId: `planet-${slug}`,
      facts,
      judgments,
      persona,
      canon: {
        nakshatra: dossier.nakshatraRecord,
        manzil: dossier.manzilRecord,
        decan: dossier.decanRecord
          ? {
              number: dossier.decanRecord.decanNumber,
              elementalFamilyRuler: dossier.decanRecord.triplicitySubRuler,
            }
          : null,
      },
    };
  });
}


/**
 * Reuse the exact placement association logic for whole-chart interpretation.
 * Reference the full ledger rather than duplicating or truncating judgments.
 * Unmatched findings remain in the complete dossier, not excluded from it.
 */
export function buildWholeChartPlacementIndex(chart: ChartResult, synthesis: SynthesisResult) {
  const packets = buildPlacementPackets(chart, synthesis, { requireJudgments: false });
  const linked = new Set(packets.flatMap(p => p.judgments.map(j => j.findingId)));
  return {
    placements: packets.map(p => ({
      body: p.name,
      position: p.facts[0].value,
      findingIds: Array.from(new Set(p.judgments.map(j => j.findingId))),
      methodIds: Array.from(new Set(p.judgments.map(j => j.methodId))),
      canon: p.canon,
    })),
    chartWideOrUnlinkedFindingIds: synthesis.methodOutcomes.flatMap(m =>
      m.findingJudgments.filter(f => !linked.has(f.id)).map(f => f.id)
    ),
  };
}

/** Retain every relevant judgment once; repeated doctrine is interned losslessly. */
export function packPlacementMiniPrompt(
  packets: ReturnType<typeof buildPlacementPackets>
) {
  const unique = Array.from(
    new Map(
      packets.flatMap(p => p.judgments).map(j => [j.findingId, j])
    ).values()
  );
  const split = (text: string) => text.split(/(?<=[.!?])(?=\s)/);
  const counts = new Map<string, number>();
  for (const j of unique)
    for (const part of split(j.judgment))
      counts.set(part, (counts.get(part) ?? 0) + 1);
  const maxReferenceLength = String(counts.size).length;
  const textTable = Array.from(counts)
    .filter(([part, count]) => {
      const literalLength = JSON.stringify(part).length;
      return (count - 1) * literalLength > count * maxReferenceLength + 1;
    })
    .map(([part]) => part);
  const textIndex = new Map(textTable.map((part, i) => [part, i]));
  const sources = Array.from(new Set(unique.flatMap(j => j.sources)));
  const findingIndex = new Map(unique.map((j, i) => [j.findingId, i]));
  return {
    textTable,
    sources,
    findingColumns: [
      "judgmentParts",
      "methodId",
      "evidenceIds",
      "frame",
      "sourceRefs",
      "limitation",
    ],
    findings: unique.map(j => [
      split(j.judgment).map(part => textIndex.get(part) ?? part),
      j.methodId,
      j.evidenceIds,
      j.frame,
      j.sources.map(s => sources.indexOf(s)),
      j.limitation ?? null,
    ]),
    placements: packets.map(p => ({
      name: p.name,
      position: p.facts[0].value,
      placementEvidenceId: p.placementEvidenceId,
      facts: p.facts.map(f => ({ id: f.id, layerId: f.layerId })),
      findingRefs: p.judgments.map(j => findingIndex.get(j.findingId)),
    })),
  };
}

/** One compact outcome per relevant method; full judgments remain in the evidence drawer. */
export function boundedPlacementPacket(
  packets: ReturnType<typeof buildPlacementPackets>,
  limit = 260
) {
  return packets.map(p => {
    const methods = Array.from(new Set(p.judgments.map(j => j.methodId)));
    return {
      name: p.name,
      position: p.facts[0].value,
      placementEvidenceId: p.placementEvidenceId,
      methods: methods.map(methodId => {
        const matching = p.judgments.filter(j => j.methodId === methodId);
        const selected =
          matching.find(j => j.mechanism.includes(p.name)) ?? matching[0];
        // Exact excerpts from executed rules; never a generic invented replacement.
        const statement =
          selected.judgment.length <= limit
            ? selected.judgment
            : selected.judgment.slice(0, limit) + "…";
        return {
          methodId,
          judgment: statement,
          evidenceIds: selected.evidenceIds.slice(0, 2),
          limitation: selected.limitation,
        };
      }),
    };
  });
}

export function validatePlacementSummaries(
  value: unknown,
  packets: ReturnType<typeof buildPlacementPackets>
) {
  const result = summarySchema.parse(value);
  const expected = new Map(packets.map(p => [p.name, p]));
  if (
    result.placements.length !== expected.size ||
    new Set(result.placements.map(p => p.name)).size !== expected.size
  ) {
    throw new Error(
      "PLACEMENT_SUMMARY_INCOMPLETE: A paragraph is required for each displayed planet."
    );
  }
  for (const placement of result.placements) {
    const packet = expected.get(placement.name);
    if (!packet)
      throw new Error("PLACEMENT_SUMMARY_UNGROUNDED: Unknown planet.");
    const ids = new Set(packet.facts.map(f => f.id));
    const layers = new Set(packet.facts.map(f => f.layerId));
    if (
      placement.evidenceIds.some(id => !ids.has(id)) ||
      placement.layerIds.some(id => !layers.has(id))
    ) {
      throw new Error(
        `PLACEMENT_SUMMARY_UNGROUNDED: ${placement.name} cites unavailable evidence.`
      );
    }
    if (
      placement.paragraph.trim().split(/\s+/).length > 140 ||
      /\n\s*\n/.test(placement.paragraph)
    ) {
      throw new Error(
        `PLACEMENT_SUMMARY_TOO_LONG: ${placement.name} must fit one short paragraph.`
      );
    }
    // Sign/house boilerplate alone cannot pass as a synthesized placement paragraph.
    for (const layer of [
      "vedic-nakshatras-and-padas",
      "arabic-lunar-manazil",
      "elemental-decans",
    ]) {
      if (
        layers.has(layer) &&
        (!placement.layerIds.includes(layer) ||
          !packet.facts.some(
            f => f.layerId === layer && placement.evidenceIds.includes(f.id)
          ))
      ) {
        throw new Error(
          `PLACEMENT_SUMMARY_SHALLOW: ${placement.name} omits its calculated mansion layers.`
        );
      }
    }
    if (!placement.evidenceIds.includes(packet.placementEvidenceId)) {
      throw new Error(
        `PLACEMENT_SUMMARY_UNGROUNDED: ${placement.name} lacks its own placement.`
      );
    }
  }
  return result;
}

export async function buildChartPageReading(
  chart: ChartResult,
  mode: "natal" | "transit" | "combined",
  names: string[] = ["Sun", "Moon"]
): Promise<ChartPageReading> {
  const verified = await synthesizeChartForReading(
    chart,
    mode === "transit" ? "combined" : mode,
    "Compact individual placement card summaries"
  );
  const event = verified.snapshot.transit?.framework;
  if (mode === "transit" && !event)
    throw new Error("TRANSIT_FRAMEWORK_MISSING: Refresh the event chart.");
  const { synthesis, canonicalChart } =
    mode === "transit"
      ? await synthesizeMasterSnapshot(
          event!,
          "natal",
          "Compact event placement card summaries"
        )
      : verified;
  if (!names.length || names.length > 2 || new Set(names).size !== names.length)
    throw new Error("PLACEMENT_MINI_SCOPE: Request one or two placements.");
  const allPackets = buildPlacementPackets(canonicalChart, synthesis);
  const packets = names.map(name => {
    const packet = allPackets.find(p => p.name === name);
    if (!packet)
      throw new Error(`PLACEMENT_MINI_SCOPE: Unknown placement ${name}`);
    return packet;
  });
  const key = createHash("sha256")
    .update(
      placementMiniIdentity(chart, mode === "transit" ? "transit" : "natal") +
        JSON.stringify([mode, synthesis.snapshotId, names])
    )
    .digest("hex");
  const cached = placementCache.get(key);
  if (cached && cached.expiresAt > Date.now()) return cached.promise;
  const promise = generatePlacementParagraphs(packets, synthesis);
  placementCache.set(key, { expiresAt: Date.now() + 30 * 60_000, promise });
  while (placementCache.size > 32)
    placementCache.delete(placementCache.keys().next().value!);
  try {
    return await promise;
  } catch (error) {
    if (placementCache.get(key)?.promise === promise)
      placementCache.delete(key);
    throw error;
  }
}

async function generatePlacementParagraphs(
  packets: ReturnType<typeof buildPlacementPackets>,
  synthesis: SynthesisResult
): Promise<ChartPageReading> {
  const instruction = buildAstrologyMiniInterpreterSystem(
    `Write only the requested placement readings in candid, plainspoken, intimate second-person prose ("you / your"). Read the entire calculated packet silently and weave converging layers into one grounded everyday insight, giving natural gifts the same weight as difficult patterns.
Return JSON {"placements":[{"name":"Sun","paragraph":"...","evidenceIds":["..."],"layerIds":["..."]}]}.
REQUIREMENTS:
1. Keep each "paragraph" to one flowing paragraph of 70–110 words (max 140 words). Vary opening rhythm.
2. Start with what the reader actually does in ordinary life—what they are trying to protect or get, what comes naturally that people rely on them for, and how their behavior lands with other people.
3. Keep all astrology terminology out of "paragraph": never name planets, zodiac signs, houses, degrees, aspects, lunar mansions or stations, bounds, faces, or Lots in reader-facing text. Put all technical citations strictly inside "evidenceIds" and "layerIds".
4. Never use bullet points, headings, boilerplate openings, stock transitions, generic encouragement, recap endings, or invented events/diagnoses/motives.`
  );
  const requestFor = (limit: number) => ({
    max_tokens: 2000,
    // Keep comfortably under Render reverse proxy gateway timeouts (which abort at 60s-100s).
    timeoutMs: 25_000,
    response_format: {
      type: "json_schema" as const,
      json_schema: {
        name: "placement_minis",
        strict: true,
        schema: {
          type: "object",
          properties: {
            placements: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  name: { type: "string", enum: packets.map(p => p.name) },
                  paragraph: { type: "string" },
                  evidenceIds: { type: "array", items: { type: "string" } },
                  layerIds: { type: "array", items: { type: "string" } },
                },
                required: ["name", "paragraph", "evidenceIds", "layerIds"],
                additionalProperties: false,
              },
            },
          },
          required: ["placements"],
          additionalProperties: false,
        },
      },
    },
    messages: [
      { role: "system" as const, content: instruction },
      {
        role: "user" as const,
        content: JSON.stringify({
          chartScope: synthesis.methodOutcomes[0]?.chartScope,
          placements: boundedPlacementPacket(packets, limit),
        }),
      },
    ],
  });
  // Retain all method IDs/evidence while tightening individual excerpts when
  // new Canon layers increase the packet. Split only if even minimum excerpts
  // cannot fit; never silently remove a method or invent substitute judgments.
  let request = requestFor(260);
  for (const excerptLimit of [150, 120, 95, 75, 50]) {
    if (JSON.stringify(request.messages).length <= MINI_REQUEST_CHAR_LIMIT) break;
    request = requestFor(excerptLimit);
  }
  if (
    JSON.stringify(request.messages).length > MINI_REQUEST_CHAR_LIMIT &&
    packets.length > 1
  ) {
    const first = await generatePlacementParagraphs([packets[0]], synthesis);
    const second = await generatePlacementParagraphs([packets[1]], synthesis);
    return {
      snapshotId: synthesis.snapshotId,
      dominantConclusion: synthesis.dominantConclusion,
      methodOutcomes: synthesis.methodOutcomes,
      themes: synthesis.themes,
      methodLedgerSummary: synthesis.methodLedgerSummary,
      qualityAudit: synthesis.qualityAudit,
      placements: [...first.placements, ...second.placements],
    };
  }
  if (JSON.stringify(request.messages).length > MINI_REQUEST_CHAR_LIMIT)
    throw new Error(
      "PLACEMENT_MINI_BUDGET: This placement exceeds the compact request budget."
    );
  const result = await runInterpretation("PLACEMENT_SUMMARY", synthesis, request);
  const raw = JSON.parse(interpretationText(result).replace(/^\s*```(?:json)?\s*/, "").replace(/\s*```\s*$/, ""));
  if (raw && Array.isArray(raw.placements)) {
    raw.placements = raw.placements.map((p: any) => ({
      ...p,
      paragraph: typeof p?.paragraph === "string" ? scrubNonHoraryAstrologyJargon(p.paragraph) : p?.paragraph,
    }));
  }
  const reading = validatePlacementSummaries(raw, packets);
  return {
    snapshotId: synthesis.snapshotId,
    dominantConclusion: synthesis.dominantConclusion,
    methodOutcomes: synthesis.methodOutcomes,
    themes: synthesis.themes,
    methodLedgerSummary: synthesis.methodLedgerSummary,
    qualityAudit: synthesis.qualityAudit,
    placements: reading.placements.map(p => ({
      ...p,
      evidence: packets
        .find(packet => packet.name === p.name)!
        .facts.filter(f => p.evidenceIds.includes(f.id)),
    })),
  };
}

/**
 * Calculates accurate chronological age from birth date string to any reference moment.
 */
export function calculatePersonAge(birthDateStr: string, refDate: Date = new Date()): number {
  const birth = new Date(birthDateStr);
  if (isNaN(birth.getTime())) return 30; // fallback default
  const diffMs = refDate.getTime() - birth.getTime();
  const ageYears = diffMs / (365.2425 * 86400000);
  return Math.max(0, Number(ageYears.toFixed(2)));
}

/**
 * Computes calendar date from birth date and fractional years elapsed.
 */
export function addYearsToDate(birthDateStr: string, yearsElapsed: number): string {
  const birth = new Date(birthDateStr);
  if (isNaN(birth.getTime())) return new Date().toISOString().slice(0, 10);
  const targetMs = birth.getTime() + yearsElapsed * 365.2425 * 86400000;
  return new Date(targetMs).toISOString().slice(0, 10);
}

    const profectedSignIdx = (ascSignIdx + (boundedAge % 12)) % 12;
    profectedSignAgent = ZODIAC_SIGNS[profectedSignIdx];
    lordOfTheYearAgent = TRADITIONAL_RULERS[profectedSignAgent] || "Mars";
    profectedHouseAgent = (boundedAge % 12) + 1;
  }

  const traditionalThemes = HOUSE_THEMES_HELLENISTIC[profectedHouseAgent] || [];

  return {
    age: boundedAge,
    yearNumber,
    profectedHouseGod,
    profectedHouseAgent,
    profectedSignGod,
    profectedSignAgent,
    lordOfTheYearGod,
    lordOfTheYearAgent,
    traditionalThemes,
  };
}

export interface LifeStoryBeat {
  period: string;
  lifeDomain: string;
  configuration: string;
  manifestation: string;
  consequence: string;
  timingBasis: string;
  findingIds: string[];
  counterEvidenceIds: string[];
  prose: string;
}

export interface LifeStoryPlan {
  beats: LifeStoryBeat[];
  gaps: string[];
}

export function parseLifeStoryPlan(
  raw: unknown,
  methods: readonly import("./astrologyKnowledge/synthesis/types").MethodOutcome[]
): LifeStoryPlan {
  const validIds = new Set(methods.flatMap(method => [
    ...method.calculatedFindings.map(finding => finding.evidenceId),
    ...method.findingJudgments.flatMap(finding => [finding.id, ...finding.evidenceIds]),
  ]));
  const asRecord = (value: unknown): Record<string, unknown> =>
    value !== null && typeof value === "object" && !Array.isArray(value)
      ? value as Record<string, unknown> : {};
  const asText = (value: unknown) => typeof value === "string" ? value.trim() : "";
  const asStrings = (value: unknown): string[] =>
    Array.isArray(value) ? value.filter((v): v is string => typeof v === "string").map(v => v.trim()).filter(Boolean) : [];
  const source = asRecord(raw);
  const gaps = asStrings(source.gaps);
  const beats: LifeStoryBeat[] = [];
  for (const value of Array.isArray(source.beats) ? source.beats : []) {
    const item = asRecord(value);
    const requestedIds = asStrings(item.findingIds);
    const findingIds = Array.from(new Set(requestedIds.filter(id => validIds.has(id))));
    const requestedCounterIds = asStrings(item.counterEvidenceIds);
    const counterEvidenceIds = Array.from(new Set(requestedCounterIds.filter(id => validIds.has(id))));
    if (requestedIds.some(id => !validIds.has(id)) || requestedCounterIds.some(id => !validIds.has(id)))
      gaps.push("A generated evidence reference was not present in the delivered chart dossier.");
    const beat: LifeStoryBeat = {
      period: asText(item.period), lifeDomain: asText(item.lifeDomain),
      configuration: asText(item.configuration), manifestation: asText(item.manifestation),
      consequence: asText(item.consequence), timingBasis: asText(item.timingBasis),
      findingIds, counterEvidenceIds,
      prose: scrubNonHoraryAstrologyJargon(asText(item.prose)),
    };
    if (!beat.period || !beat.lifeDomain || !beat.configuration || !beat.manifestation ||
        !beat.consequence || !beat.timingBasis || !beat.prose || !beat.findingIds.length)
      throw new Error("LIFE_STORY_PLAN_INCOMPLETE: A proposed life development lacked its required chart reference or narrative. Retry this chapter.");
    beats.push(beat);
  }
  if (!beats.length) throw new Error("LIFE_STORY_PLAN_EMPTY: No evidence-linked life developments were returned. Retry this chapter.");
  // Exact IDs establish traceability, not empirical proof or semantic entailment.
  return { beats, gaps: Array.from(new Set(gaps)) };
}

function truncateEvidenceText(text: unknown, maxChars: number): string {
  const raw = typeof text === "string" ? text : text == null ? "" : String(text);
  const clean = raw.replace(/\s+/g, " ").trim();
  if (clean.length <= maxChars) return clean;
  const slice = clean.slice(0, maxChars - 1);
  const stop = Math.max(slice.lastIndexOf(". "), slice.lastIndexOf("; "), slice.lastIndexOf(" "));
  return (stop > maxChars * 0.55 ? slice.slice(0, stop) : slice).trim() + "…";
}

function compactCalculationResults(value: unknown, depth = 0): unknown {
  if (value === null || typeof value === "number" || typeof value === "boolean") return value;
  if (typeof value === "string") return truncateEvidenceText(value, 180);
  if (Array.isArray(value)) {
    if (depth >= 2) return `[${value.length} items]`;
    const limit = depth === 0 ? 16 : 8;
    const head = value.slice(0, limit).map(item => compactCalculationResults(item, depth + 1));
    return value.length > limit ? [...head, `…(+${value.length - limit})`] : head;
  }
  if (typeof value === "object") {
    if (depth >= 3) return "[summary]";
    const entries = Object.entries(value as Record<string, unknown>);
    const limit = depth === 0 ? 18 : 12;
    const out: Record<string, unknown> = {};
    for (const [k, v] of entries.slice(0, limit)) {
      out[k] = compactCalculationResults(v, depth + 1);
    }
    if (entries.length > limit) {
      out._more = entries.length - limit;
    }
    return out;
  }
  return value;
}

function focusedPlacementEvidence(synthesis: SynthesisResult, focus: string) {
  // A contact has two bodies. Keep both sides' method findings while
  // excluding unrelated chart-wide content from the provider payload.
  // The separator is internal and comes from canonical chart body names.
  const bodyPatterns = focus.split("|").map(name => name.trim()).filter(Boolean)
    .map(name => new RegExp(`(?:^|[^a-z])${name.toLowerCase().replaceAll(" ", "[-_ ]")}(?:$|[^a-z])`, "i"));
  const matchesBody = (value: string) => bodyPatterns.some(pattern => pattern.test(value));
  // Complete method coverage is not the same as resending every unrelated
  // paragraph. This is a focused *reading* packet, not a raw chart archive.
  // Keep every method/status and explicit omission counts when a method's
  // selected details exceed its share of the model request budget.
  const perMethodBudget = Math.max(210, Math.floor(10_000 / Math.max(1, synthesis.methodOutcomes.length)));
  const short = (value: unknown, limit: number) => {
    const text = typeof value === "string" ? value : value == null ? "" : String(value);
    return text.length > limit ? text.slice(0, limit) + "…" : text;
  };
  let omittedJudgments = 0;
  let omittedCalculations = 0;
  const methods = synthesis.methodOutcomes.map(method => {
    const namedCalculations = new Set(method.calculatedFindings
      .filter(finding => matchesBody(finding.label) || matchesBody(finding.evidenceId))
      .map(finding => finding.evidenceId));
    const judgments = method.findingJudgments.filter(finding =>
      matchesBody(finding.subject) ||
      finding.evidenceIds.some(id => matchesBody(id) || namedCalculations.has(id))
    );
    const citedIds = new Set(judgments.flatMap(finding => finding.evidenceIds));
    const calculations = method.calculatedFindings.filter(finding =>
      namedCalculations.has(finding.evidenceId) || citedIds.has(finding.evidenceId)
    );
    const selectedJudgments: unknown[] = [];
    const selectedCalculations: unknown[] = [];
    let used = 0;
    const add = (destination: unknown[], value: unknown) => {
      const size = JSON.stringify(value).length;
      if (used + size > perMethodBudget) return false;
      destination.push(value);
      used += size;
      return true;
    };
    // Direct, calculated judgments take precedence over long nested objects.
    for (const finding of judgments) {
      const item = [
        finding.id, finding.subject,
        short(finding.judgment, 240), short(finding.mechanism, 160),
        finding.evidenceIds, finding.limitation ?? null,
      ];
      if (!add(selectedJudgments, item)) omittedJudgments++;
    }
    for (const finding of calculations) {
      const item = [
        finding.evidenceId, finding.label,
        short(finding.calculatedValue, 220),
      ];
      if (!add(selectedCalculations, item)) omittedCalculations++;
    }
    return [
      method.methodId, method.methodName, method.status, method.frame,
      selectedJudgments, selectedCalculations,
      { selected: [selectedJudgments.length, selectedCalculations.length],
        omitted: [judgments.length - selectedJudgments.length, calculations.length - selectedCalculations.length] },
    ];
  });
  return {
    snapshotId: synthesis.snapshotId,
    focus,
    coverage: {
      methodsTotal: synthesis.methodOutcomes.length,
      methodsIncluded: methods.length,
      omittedJudgments,
      omittedCalculations,
      detailsCompacted: true,
      note: "Every method and status is included. Only focus-linked judgments and calculations are summarized. Omitted detail counts are explicit; full calculated evidence remains available in the technical ledger.",
    },
    methodColumns: ["id", "name", "status", "frame", "focusedJudgments", "focusedCalculations", "detailCounts"],
    judgmentColumns: ["id", "subject", "judgment", "mechanism", "evidenceIds", "limitation"],
    focusedCalculationColumns: ["id", "label", "calculatedValue"],
    methods,
    calculationColumns: ["id", "executionStatus"],
    calculations: synthesis.sourceSnapshot.methodExecutionLedger.map(method => [method.id, method.status]),
  };
}

export function interpretationText(
  response: Awaited<ReturnType<typeof invokeLLM>>
) {
  const content = response.choices?.[0]?.message?.content;
  return (
    typeof content === "string"
      ? content
      : (content ?? [])
          .filter(part => part.type === "text")
          .map(part => part.text)
          .join("")
  ).trim();
}

export function parseLlmJsonObject(
  rawText: string,
  errorCode = "INTERPRETATION_INVALID_JSON"
): unknown {
  const cleaned = rawText
    .trim()
    .replace(/^\s*```(?:json)?\s*/i, "")
    .replace(/\s*```\s*$/i, "")
    .trim();
  if (!cleaned) {
    throw new Error(`${errorCode}: Provider returned empty structured output.`);
  }
  try {
    return JSON.parse(cleaned);
  } catch {
    const start = cleaned.indexOf("{");
    const end = cleaned.lastIndexOf("}");
    if (start >= 0 && end > start) {
      try {
        return JSON.parse(cleaned.slice(start, end + 1));
      } catch {
        // Fall through to contextual error below
      }
    }
    throw new Error(
      `${errorCode}: Provider returned unreadable structured output.`
    );
  }
}

export async function runInterpretation(
  route: string,
  synthesis: SynthesisResult | readonly SynthesisResult[],
  request: Request,
  layout: "complete" | "placement-reference" | "horary-compact" | { focusedPlacement: string } = "complete"
) {
  const ledgers = (
    Array.isArray(synthesis) ? synthesis : [synthesis]
  ) as readonly SynthesisResult[];
  if (!ledgers.length)
    throw new Error(
      `${route}_EVIDENCE_MISSING: No calculated synthesis was provided.`
    );
  const messages = [...request.messages];
  const firstUser = messages.findIndex(message => message.role === "user");
  if (layout === "placement-reference") {
    const placement = messages[firstUser];
    if (
      ledgers.length !== 1 ||
      !placement ||
      typeof placement.content !== "string"
    )
      throw new Error(
        "PLACEMENT_SYNTHESIS_EVIDENCE_MISSING: Expected one placement packet."
      );
    messages[firstUser] = {
      ...placement,
      content:
        placement.content +
        "\n\nFULL CALCULATED CHART AND METHOD OUTCOMES:\n" +
        stringifyEvidence(interpretationEvidence(ledgers[0]))
    };
  } else {
    const sharedContract = {
      role: "system" as const,
      content: "UNIVERSAL EVIDENCE-WEAVING CONTRACT: Read all applicable calculated methods and their statuses. Do not list parallel methods as a substitute for interpretation. Explain their convergence, contradictions and implications without invented facts. Preserve route-specific Horary verdicts, JSON schemas and evidence IDs.",
    };
    const wholeChart = route === "MASTER_READING" || route === "LIFE_STORY";
    const evidence =
      typeof layout === "object"
        ? stringifyEvidence(ledgers.map(ledger => focusedPlacementEvidence(ledger, layout.focusedPlacement)))
        : layout === "horary-compact"
          ? stringifyEvidence(ledgers.map(horaryInterpretationEvidence))
          : stringifyEvidence(ledgers.map(ledger => ({
              ...interpretationEvidence(ledger),
              ...(wholeChart ? {
                placementIndex: ledger.sourceSnapshot.natal
                  ? buildWholeChartPlacementIndex(ledger.sourceSnapshot.natal, ledger)
                  : { status: "not_calculated", reason: "Canonical natal chart unavailable" },
              } : {}),
            })));
    const dossier = {
      role: "user" as const,
      content:
        (typeof layout === "object"
          ? "FOCUSED PLACEMENT DOSSIER. Read every method row. Apply chart-wide context only when it bears directly on the selected placement. Treat missing findings and method statuses as limits on interpretation.\n"
          : layout === "horary-compact"
            ? "HORARY CALCULATED METHOD DOSSIER. Every registered method and status is present. The specific Horary judgment testimony is in the preceding user message. Findings are budgeted and exact omitted counts are supplied; never invent evidence omitted for size.\n"
            : "COMPLETE CALCULATED INTERPRETATION DOSSIER. Consider every applicable method, connecting agreement and tension to the specific question. Read each table using its matching column names. canonRefs index the canon table. Respect frames, evidence IDs and method status. Never fill gaps from inference. Previous prose is conversation context only.\n") +
        (wholeChart ? "\nSHARED PLANETARY EVIDENCE INDEX: Use placementIndex to follow each planet into the exact finding IDs in the full method tables. Connect different planets, occupied and ruled houses, dignities, aspects, mansions and other methods into one configuration-level conclusion. A planet may participate in multiple distinct configurations. Examine chartWideOrUnlinkedFindingIds too: every finding remains in the full dossier. Do not write a series of isolated placement readings or simply paraphrase a domain label.\n" : "") +
        evidence,
    };
    // Preserve the domain packet and keep the current turn last in conversations.
    const index =
      firstUser < 0
        ? messages.length
        : firstUser === messages.length - 1
          ? firstUser + 1
          : messages.length - 1;
    messages.splice(index, 0, sharedContract, dossier);
  }
  // Apply the behavior contract to every route, including placement-reference,
  // while avoiding duplicate instructions in routes using the Master Interpreter.
  const alreadyUsesReadingBehavior = messages.some(
    message =>
      message.role === "system" &&
      typeof message.content === "string" &&
      message.content.includes("THE ABSOLUTE — AS REAL AS IT GETS: SOUL-DEEP READING-BEHAVIOR CONTRACT")
  );
  // Life Story is a distinct chronological biography, not the Who Are You
  // identity portrait. Share knowledge, not conflicting reading instructions.
  const isLifeStory = route === "LIFE_STORY";
  if (isLifeStory || !alreadyUsesReadingBehavior) {
    const firstNonSystem = messages.findIndex(m => m.role !== "system");
    const insertAt = firstNonSystem < 0 ? messages.length : firstNonSystem;
    messages.splice(insertAt, 0, {
      role: "system",
      content: isLifeStory
        ? [
            COMBINATION_INTELLIGENCE_CANON,
            "LIFE STORY — CHRONOLOGICAL BIOGRAPHY AND DEPTH CONTRACT",
            "The dedicated Life Historian instructions control this route. Use the Genesis inquiry, complete supplied chart testimony and stage-specific timing to construct a distinct life chapter, not a psychological portrait. The Canon is shared with Who Are You, but its application here is to changing worldly circumstances across time.",
            "Aim for 6–9 substantial, connected paragraphs (roughly 600–1000 words where the evidence supports that depth). Establish the stage's opening circumstances, develop the strongest connected life domains, show turning points and their consequences, and close on the transition to the next era. Do not stretch weak evidence or repeat a central personality label just to meet length.",
            "Give supported achievements, opportunities, relationships, family responsibilities, education, work, material conditions, setbacks, recoveries and contribution their rightful place. Select the domains from the chart rather than forcing a fixed life script. Use distinct configurations and cross-method testimony to explain each development; do not manufacture dates, events, diagnoses or actual biographical knowledge.",
            "Write in direct second person with profound, accessible, literary human language. Avoid clinical terminology, generic motivational advice, repetitive abstractions, and canned moral endings. Keep technical evidence in the evidence view. Preserve uncertainty for unverified past events and future possibilities, and distinguish supplied timing from symbolic inference.",
          ].join("\n\n")
        : READING_BEHAVIOR_DIRECTIVE,
    });
  }
  const response = await invokeLLM({
    ...request,
    ...(isLifeStory && request.max_tokens == null && request.maxTokens == null
      ? { max_tokens: 6000 }
      : {}),
    messages,
  });
  const finish = response.choices?.[0]?.finish_reason;
  if (
    finish &&
    !["stop", "end_turn", "stop_sequence", "completed"].includes(
      finish.toLowerCase()
    )
  )
    throw new Error(
      `${route}_INCOMPLETE: The provider truncated or stopped the interpretation (${finish}).`
    );
  if (!interpretationText(response))
    throw new Error(
      `${route}_EMPTY: The interpretation service returned no reading.`
    );
  return response;
}

}

async function generate(synthesis: SynthesisResult, question: string, supplementalEvidence: string) {
  const canonicalChart = synthesis.sourceSnapshot.natal;
  let personaMatrix: PlanetPersonaTrait[] = [];
  let personaMatrixMarkdown = "";
  let canonicalMappings: CanonicalPlacementMapping[] = [];

  const requestedActors = extractRequestedActors(question);

  if (canonicalChart) {
    try {
      personaMatrix = buildPlanetPersonaMatrix(canonicalChart);
      canonicalMappings = mapRequestedPlacementsToCanonicalMeaning(canonicalChart, requestedActors);
      const dossier = retrieveChartCanonDossier(canonicalChart);
      const ruledMap = getHousesRuledByPlanet(canonicalChart);

      const ruledSummary = Array.from(ruledMap.entries())
        .map(([planet, houses]) => `- ${planet} rules House${houses.length > 1 ? "s" : ""} ${houses.join(", ")}`)
        .join("\n");

      const lotsSummary = dossier.arabicLotsSummary
        .map(l => `- ${l.lot.name}: ${l.sign}, House ${l.house} (${l.lot.psychologicalMeaning})`)
        .join("\n");

      const relationalSummary = (dossier.relationalChains ?? [])
        .map(rc => `- House ${rc.house} (${rc.houseTopic}) → ${rc.ruler} in ${rc.rulerSign} (House ${rc.rulerHouse}): ${rc.inherentDignityVsDelivery}`)
        .join("\n");

      const mechanismSummary = dossier.detectedMechanisms
        .map(m => `- ${m.title}: ${m.statement}`)
        .join("\n");

      const dispositorSummary = (dossier.dispositorChains ?? [])
        .map(d => `- ${d.planet} in ${d.sign} → host ${d.domicileLord} (${d.receptionWithLord.replace(/-/g, " ")}; chain: ${d.chainToFinalDispositor.join(" → ")})`)
        .join("\n");

      const yogasSummary = (dossier.vedicJyotishSystem?.detectedYogas ?? [])
        .map(y => `- ${y.name} (${y.status}): ${y.humanRealitySynthesis}`)
        .join("\n");

      const charaSummary = dossier.vedicJyotishSystem?.charaKarakas
        ?.map(k => `${k.karaka}: ${k.planet}`)
        .join(" | ");

      const placementByBody = new Map(dossier.placements.map(pl => [pl.body, pl]));
      const traitByBody = new Map(personaMatrix.map(p => [p.planet, p]));
      const preDetectedConfigs = buildNamedChartConfigurations(personaMatrix, ruledMap);

      personaMatrixMarkdown = "--- PLANET PERSONA CANON MATRIX (ALL CALCULATED LAYERS) ---\n" +
        "| Planet | Sign & Deg | Agent & God Frame | Chaldean Face, Triplicity & Bounds | Nakshatra, Pada & D9 | Manzil | Dignity, Sect & Capacity | Aspects |\n" +
        "|---|---|---|---|---|---|---|---|\n" +
        personaMatrix.map(p => {
          const ruled = ruledMap.get(p.planet) ?? [];
          const ruledLabel = ruled.length ? ` (Rules H${ruled.join(", H")})` : "";
          const pl = placementByBody.get(p.planet);
          const godFrameLabel = pl?.godHouse && pl.godHouse !== p.house ? ` [God H${pl.godHouse} archetype]` : "";
          const subLayersLabel = `Chaldean Face: ${p.chaldeanFaceRuler}; Triplicity Sub-ruler: ${p.decanSubRuler}; ${p.boundsEgyptian}`;
          const nakDetail = pl?.nakshatraRecord ? `${p.nakshatraName} P${p.nakshatraPada} (${p.nakshatraDeity}; ${pl.nakshatraRecord.shakti})` : `${p.nakshatraName} P${p.nakshatraPada} (${p.nakshatraDeity})`;
          return `| ${p.planet} | ${p.formattedPosition} | Agent H${p.house}${ruledLabel}${godFrameLabel}: ${p.houseArena.slice(0, 36)} | ${subLayersLabel} | ${nakDetail} / D9 ${p.navamshaD9} | ${p.manzilName} | ${p.sovereigntyTier} / ${p.shadbalaStatus} (${p.houseSupportTier}) | ${p.activeAspects.slice(0, 3).join("; ") || "Solo"} |`;
        }).join("\n") + "\n\n" +
        (preDetectedConfigs.length ? `--- PRE-DETECTED MULTI-PLANET CONFIGURATIONS ---\n${preDetectedConfigs.join("\n")}\n\n` : "") +
        (ruledSummary ? `--- HOUSE RULERSHIP CHAINS ---\n${ruledSummary}\n\n` : "") +
        (dispositorSummary ? `--- DISPOSITOR CHAINS & HOST HOSPITALITY ---\n${dispositorSummary}\n\n` : "") +
        (relationalSummary ? `--- KEY RELATIONAL CHAINS ---\n${relationalSummary}\n\n` : "") +
        (lotsSummary ? `--- HERMETIC & ARABIC LOTS ---\n${lotsSummary}\n\n` : "") +
        (mechanismSummary ? `--- DETECTED TRADITIONAL MECHANISMS ---\n${mechanismSummary}\n\n` : "") +
        (dossier.arabicPersianSystem?.almutenFiguris ? `--- ALMUTEN FIGURIS & FIRDARIA ---\nAlmuten Winner: ${dossier.arabicPersianSystem.almutenFiguris.winnerPlanet}. ${dossier.arabicPersianSystem.firdariaSystem.explanation}\n\n` : "") +
        (yogasSummary || charaSummary ? `--- VEDIC YOGAS & CHARA KARAKAS ---\n${charaSummary ? `Karakas: ${charaSummary}\n` : ""}${yogasSummary}\n\n` : "") +
        "--- PRE-NARRATIVE CANONICAL MAPPINGS (MANDATORY REQUESTED ACTORS) ---\n" +
        canonicalMappings.map(m => {
          const trait = traitByBody.get(m.actor) ?? personaMatrix.find(p => p.planet.replace(/\s+/g, "").toLowerCase() === m.actor.replace(/\s+/g, "").toLowerCase());
          const subLayerLine = trait
            ? `Chaldean Face: ${trait.chaldeanFaceRuler}; Triplicity Sub-ruler: ${trait.decanSubRuler}; ${m.boundsEgyptian}`
            : m.boundsEgyptian;
          return (
            `[ACTOR: ${m.actor} (${m.symbol}) in ${m.sign}, House ${m.house}]\n` +
            `- Life Arena: ${m.houseArena}\n` +
            `- Sub-Sign Layers (Keep Strictly Distinct): ${subLayerLine}\n` +
            `- Nakshatra & Manzil: ${m.nakshatraName} Pada ${m.nakshatraPada} (${m.nakshatraDeity}; ${m.nakshatraShakti}); ${m.manzilName}\n` +
            `- Dignity & Chart Weight: ${m.sovereigntyTier}; ${m.shadbalaStatus}; House support: ${m.houseSupportTier}\n` +
            `- Real-World Meaning: ${m.realWorldTranslation.coreTheme}\n` +
            `- Behavioral Guide: ${m.realWorldTranslation.conversationalProse}`
          );
        }).join("\n\n") + "\n\n";
    } catch {
      // Preserve the calculated dossier and record the missing prompt section.
      console.warn("[NARRATIVE_PERSONA_MATRIX_UNAVAILABLE] Could not prepare the optional display matrix.");
    }
  }

  const response = await runInterpretation("MASTER_READING", synthesis, {
      max_tokens: 8192,
      timeoutMs: 35_000,
      messages: [
        {
          role: "system",
          content: FULL_SYNTHESIS_STYLE + `\n\n${SYNTHESIS_LAYERS_AND_EXAMPLES}\n\nSILENT SYNTHESIS GUARDRAILS:
- Read the entire supplied chart JSON silently. Do not recalculate placements, fill in missing data, or invent chart evidence. If a field is missing or marked "not_calculated", continue with the available evidence.
- Do not repeat the same conclusion. A planet can participate in several distinct configurations when its rulership, contacts or other calculated relationships change the conclusion. Weave converging findings into integrated insights rather than echoing parallel placement descriptions.
- Return only the reading: short, bold section titles specific to this reader and topic; flowing paragraphs with varied length, rhythm, and openings; zero boilerplate openings, stock transitions, generic encouragement, or recap endings.
- Keep all astrology terminology out of the reader-facing text, including the titles. Never name planets, zodiac signs, houses, degrees, aspects, lunar mansions or stations, bounds, faces, or Lots. Keep citations and technical metadata in the Evidence tab. Use no bullet-point lists of traits. End with a sentence that belongs to this reader and this chart.`
        },
        {
          role: "user",
          content: `${personaMatrixMarkdown}${supplementalEvidence ? supplementalEvidence + "\n\n" : ""}REQUESTED TOPIC / INQUIRY: "${question}"\n\nWrite an intimate, candid, second-person portrait answering this inquiry directly using the structured chart data silently. Start with the strongest active pattern, give natural gifts the same weight as difficult patterns, make room for ordinary-life scenes across family, friendships, intimate partnerships, work, and creative interests (or follow the focused topic while pulling in related areas), use short bold section titles specific to this reader and topic, write in flowing paragraphs with no bullet-point lists of traits, and keep all astrology terminology out of both the titles and body prose. End with a single sentence that belongs to this reader and this chart.`
        },
      ],
    });

  const text = scrubNonHoraryAstrologyJargon(interpretationText(response));
  if (text.length < 80)
    throw new Error("MASTER_READING_INCOMPLETE: The provider did not return a full reading.");

  // Parse short bold standalone titles (**Title**) or legacy markdown headings (### Title)
  const rawSections = text
    .split(/(?=^(?:#{2,3}\s+[^\n]+|\*\*[^*\n]{2,120}\*\*)\s*$)/gm)
    .filter(s => s.trim().length > 0);
  const generatedSections = rawSections.map((chunk, index) => {
    const headingMatch = chunk.match(/^(?:#{2,3}\s+(?:Chapter [IVXLCDM]+:\s*)?([^\n]+)|\*\*([^*\n]{2,120})\*\*)\s*$/m);
    const title = (headingMatch?.[1] || headingMatch?.[2] || `Section ${index + 1}`).trim();
    const cleanText = scrubNonHoraryAstrologyJargon(
      chunk
        .replace(/^(?:#{2,3}\s+[^\n]+|\*\*[^*\n]{2,120}\*\*)\s*\n*/m, "")
        .replace(/^---\n+/m, "")
    );
    return {
      id: `section-${index + 1}`,
      title,
      text: cleanText || scrubNonHoraryAstrologyJargon(chunk),
      findingIds: [],
    };
  });
  const firstCleanParagraph =
    scrubNonHoraryAstrologyJargon(
      text
        .replace(/^#{1,6}\s+[^\n]+\n*/gm, "")
        .replace(/^\*\*[^*\n]{2,120}\*\*\n*/gm, "")
        .replace(/^>\s+\*\*Chart Factors:\*\*[^\n]+\n*/gm, "")
        .split(/\n\n+/)
        .map(p => p.trim())
        .find(p => p.length >= 60 && !p.startsWith("---")) ?? text
    );

  const narrative: MasterNarrative = {
    answer: text,
    sections: generatedSections.length ? generatedSections : [{ id: "complete-reading", title: "Your Complete Reading", text, findingIds: [] }],
    methodContributions: synthesis.methodOutcomes.map(outcome => ({
      methodId: outcome.methodId,
      sectionIds: generatedSections.map(s => s.id),
      findingIds: outcome.evidenceIds,
      contribution: outcome.contribution,
    })),
    methodsReviewed: synthesis.sourceSnapshot.methodExecutionLedger.map(l => l.name),
  };

  return {
    ...synthesis,
    calculatedNarrative: synthesis.integratedNarrative,
    dominantConclusion: firstCleanParagraph,
    integratedNarrative: [text],
    narrative,
    narrativeAudit: {
      passed: true,
      interpretedMethodIds: narrative.methodContributions.map(c => c.methodId),
      citedFindingIds: Array.from(new Set(narrative.sections.flatMap(s => s.findingIds))),
      validation: "direct-ai-prose" as const,
      inputCoverage: "complete-calculated-outcomes" as const,
      deliveredMethodIds: synthesis.sourceSnapshot.methodExecutionLedger.map(layer => layer.id),
      deliveredOutcomeIds: synthesis.methodOutcomes.map(outcome => outcome.methodId),
    },
  };
}

export async function narrateMasterSynthesis(synthesis: SynthesisResult, question: string, supplementalEvidence = "") {
  return generate(synthesis, question, supplementalEvidence);
}

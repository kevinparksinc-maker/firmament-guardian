const normalizeResponseFormat = ({
  responseFormat,
  response_format,
  outputSchema,
  output_schema,
}: {
  responseFormat?: ResponseFormat;
  response_format?: ResponseFormat;
  outputSchema?: OutputSchema;
  output_schema?: OutputSchema;
}):
  | { type: "json_schema"; json_schema: JsonSchema }
  | { type: "text" }
  | { type: "json_object" }
  | undefined => {
  const explicitFormat = responseFormat || response_format;
  if (explicitFormat) {
    if (
      explicitFormat.type === "json_schema" &&
      !explicitFormat.json_schema?.schema
    ) {
      throw new Error(
        "responseFormat json_schema requires a defined schema object"
      );
    }
    return explicitFormat;
  }

  const schema = outputSchema || output_schema;
  if (!schema) return undefined;

  if (!schema.name || !schema.schema) {
    throw new Error("outputSchema requires both name and schema");
  }

  return {
    type: "json_schema",
    json_schema: {
      name: schema.name,
      schema: schema.schema,
      ...(typeof schema.strict === "boolean" ? { strict: schema.strict } : {}),
    },
  };
};

type FetchInit = NonNullable<Parameters<typeof fetch>[1]>;

const sleep = (ms: number, signal?: AbortSignal | null) =>
  new Promise<void>((resolve, reject) => {
    if (signal?.aborted) return reject(signal.reason ?? new Error("LLM request aborted."));
    const onAbort = () => {
      clearTimeout(timer);
      signal?.removeEventListener("abort", onAbort);
      reject(signal?.reason ?? new Error("LLM request aborted."));
    };
    const timer = setTimeout(() => {
      signal?.removeEventListener("abort", onAbort);
      resolve();
    }, ms);
    signal?.addEventListener("abort", onAbort, { once: true });
  });

const parseRetryAfter = (value: string | null): number | undefined => {
  if (!value) return undefined;
  const seconds = Number(value);
  if (Number.isFinite(seconds)) return Math.max(0, seconds * 1000);
  const at = Date.parse(value);
  return Number.isNaN(at) ? undefined : Math.max(0, at - Date.now());
};

// Equal-jitter exponential backoff. The cap/2 floor guarantees a minimum
// delay so a misbehaving caller loop slows down instead of hammering the
// upstream while it keeps returning errors.
const computeBackoffDelay = (
  attempt: number,
  retryAfterMs?: number
): number => {
  const cap = Math.min(RETRY_BASE_DELAY_MS * 2 ** attempt, RETRY_MAX_DELAY_MS);
  const jittered = cap / 2 + Math.random() * (cap / 2);
  return Math.min(Math.max(jittered, retryAfterMs ?? 0), RETRY_MAX_DELAY_MS);
};

// Retries non-2xx responses and network errors with exponential backoff, then
// returns the final Response so callers keep their existing error handling.
const fetchWithBackoff = async (
  url: string,
  init: FetchInit & { timeoutMs?: number }
): Promise<Response> => {
  let lastError: unknown;

  for (let attempt = 0; attempt <= RETRY_MAX_RETRIES; attempt++) {
    try {
      if (init.signal?.aborted) throw init.signal.reason ?? new Error("LLM request aborted.");
      const controller = new AbortController();
      const onAbort = () => controller.abort(init.signal?.reason);
      init.signal?.addEventListener("abort", onAbort, { once: true });
      const timeoutMs = init.timeoutMs ?? LLM_REQUEST_TIMEOUT_MS;
      const timeout = setTimeout(() => controller.abort(), timeoutMs);
      let response: Response;
      try {
        const { timeoutMs: _timeoutMs, ...requestInit } = init;
        try {
          response = await fetch(url, { ...requestInit, signal: controller.signal });
        } catch (error) {
          if (controller.signal.aborted && !init.signal?.aborted) {
            throw new Error(`AI_REQUEST_TIMEOUT: The AI service did not respond within ${Math.round(timeoutMs / 1000)} seconds. Retry this summary.`, { cause: error });
          }
          throw error;
        }
      } finally {
        clearTimeout(timeout);
        init.signal?.removeEventListener("abort", onAbort);
      }
      // Preserve the original body for billing, invalid-input and auth errors.
      // Repeating a rejected 400 request hides the useful provider explanation.
      if (response.ok || ![408, 409, 429, 500, 502, 503, 504, 529].includes(response.status) || attempt === RETRY_MAX_RETRIES) {
        return response;
      }

      const retryAfterMs = parseRetryAfter(
        response.headers.get("retry-after")
      );
      try {
        await response.body?.cancel();
      } catch {
        // Body already settled; nothing to clean up.
      }
      console.warn(
        `LLM request retry ${attempt + 1}/${RETRY_MAX_RETRIES} after status ${response.status}`
      );
      await sleep(computeBackoffDelay(attempt, retryAfterMs), init.signal);
    } catch (error) {
      lastError = error;
      if (init.signal?.aborted || attempt === RETRY_MAX_RETRIES ||
          (error instanceof Error && error.message.startsWith("AI_REQUEST_TIMEOUT:"))) throw error;
      console.warn(
        `LLM request retry ${attempt + 1}/${RETRY_MAX_RETRIES} after network error`
      );
      await sleep(computeBackoffDelay(attempt), init.signal);
    }
  }

  throw lastError instanceof Error
    ? lastError
    : new Error("LLM request failed after exhausting retries");
};

      role: message.role === "assistant" ? "model" : "user",
      parts: [{ text: contentToText(message.content) }],
    }));

    responseFormat: params.responseFormat,
    response_format: params.response_format,
    outputSchema: params.outputSchema,
    output_schema: params.output_schema,
  });

  const generateWithTimeout = async () => {
    if (params.signal?.aborted) {
      throw params.signal.reason ?? new Error("LLM request aborted.");
    }
    let timer: ReturnType<typeof setTimeout> | undefined;
    let onAbort: (() => void) | undefined;
    try {
      return await Promise.race([
        ai.models.generateContent({
          model,
          contents: contents.length > 0 ? contents : "Hello",
          config: {
            maxOutputTokens: params.max_tokens ?? params.maxTokens ?? 4096,

            ...(systemInstruction ? { systemInstruction } : {}),
            ...(normalizedResponseFormat?.type === "json_object" || normalizedResponseFormat?.type === "json_schema"
              ? { responseMimeType: "application/json" }
              : {}),
          },
        }),
        new Promise<never>((_resolve, reject) => {
          timer = setTimeout(() => {
            reject(
              new Error(
                `AI_REQUEST_TIMEOUT: The AI service did not respond within ${Math.round(timeoutMs / 1000)} seconds. Retry this summary.`
              )
            );
          }, timeoutMs);
          if (params.signal) {
            onAbort = () =>
              reject(params.signal?.reason ?? new Error("LLM request aborted."));
            params.signal.addEventListener("abort", onAbort, { once: true });
          }
        }),
      ]);
    } finally {
      if (timer) clearTimeout(timer);
      if (onAbort && params.signal) params.signal.removeEventListener("abort", onAbort);
    }
  };

  let response: any;
  const maxRetries = 3;
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      response = await generateWithTimeout();
      break;
    } catch (err: any) {
      const errStr = typeof err?.message === "string" ? err.message : JSON.stringify(err);
      if (
        params.signal?.aborted ||
        errStr.startsWith("AI_REQUEST_TIMEOUT:") ||
        errStr === "LLM request aborted."
      ) {
        throw err instanceof Error ? err : new Error(errStr);
      }
      const isRateLimit =
        err?.status === "RESOURCE_EXHAUSTED" ||
        err?.status === 429 ||
        err?.code === 429 ||
        errStr.includes("429") ||
        errStr.includes("RESOURCE_EXHAUSTED") ||
        errStr.includes("Quota exceeded") ||
        errStr.includes("rate-limits");

      let retryAfterSec: number | undefined;
      const match =
        errStr.match(/retry in ([0-9.]+)s/i) ||
        errStr.match(/"retryDelay":\s*"([0-9.]+)s"/i);
      if (match && match[1]) {
        const parsed = parseFloat(match[1]);
        if (!isNaN(parsed) && parsed > 0) {
          retryAfterSec = Math.ceil(parsed);
        }
      }

      const isTransientDemand =
        err?.status === "UNAVAILABLE" ||
        err?.status === 503 ||
        err?.code === 503 ||
        errStr.includes("503") ||
        errStr.includes("UNAVAILABLE") ||
        /high demand|temporarily unavailable/i.test(errStr);

      // Only auto-retry short bursts (<= 8s) once or twice; longer cooldowns should return fast to the user.
      if ((isRateLimit || isTransientDemand) && attempt < maxRetries - 1 && (!retryAfterSec || retryAfterSec <= 8)) {
        const delayMs = retryAfterSec ? retryAfterSec * 1000 + 500 : 1500 * Math.pow(2, attempt);
        console.warn(
          `[invokeGemini] ${isRateLimit ? "Rate limit (429)" : "Transient provider load (503)"}, pausing ${delayMs}ms before retry ${attempt + 1}/${maxRetries - 1}...`
        );
        await sleep(delayMs, params.signal);
        continue;
      }

      if (isRateLimit) {
        const waitHint = retryAfterSec
          ? ` Please wait ~${retryAfterSec} seconds and try again.`
          : " Please wait ~20 seconds and try again.";
        throw new Error(
          `GEMINI_RATE_LIMIT: HTTP 429. Provider capacity was reached.${waitHint}`
        );
      }

      const statusByName: Record<string, number> = { UNAVAILABLE: 503, INVALID_ARGUMENT: 400, NOT_FOUND: 404, PERMISSION_DENIED: 403, UNAUTHENTICATED: 401 };
      const status = typeof err?.status === "number" ? err.status
        : typeof err?.code === "number" ? err.code : statusByName[err?.status];
      throw new Error(`GEMINI_REQUEST_FAILED: ${status ? `HTTP ${status}` : "Provider rejected the request"}`);
    }
  }

  const text = response?.text ?? "";
  return {
    id: `gemini-${Date.now()}`,
    created: Math.floor(Date.now() / 1000),
    model,
    choices: [
      {
        index: 0,
        message: { role: "assistant", content: text },
        finish_reason: response?.candidates?.[0]?.finishReason?.toLowerCase() ?? null,
      },
    ],
  };
}

export async function invokeLLM(params: InvokeParams): Promise<InvokeResult> {
  const chain = [
    { name: "gemini", key: ENV.geminiApiKey, invoke: invokeGemini },
    { name: "groq", key: ENV.groqApiKey, invoke: invokeGroq },
    { name: "openrouter", key: ENV.openrouterApiKey, invoke: invokeOpenRouter },
  ];
  // A provider hint selects the starting point; only configured downstream
  // providers participate. Legacy model hints never override provider models.
  const start = params.provider ? chain.findIndex(p => p.name === params.provider) : 0;
  const candidates = chain.slice(Math.max(0, start)).filter(provider => provider.key);
  const failures: string[] = [];
  for (const provider of candidates) {
    if (params.signal?.aborted) throw params.signal.reason ?? new Error("LLM request aborted.");
    const deadline = AbortSignal.timeout(params.timeoutMs ?? 60_000);
    const signal = params.signal ? AbortSignal.any([params.signal, deadline]) : deadline;
    try {
      const result = await provider.invoke({ ...params, model: undefined, signal });
      assertComplete(result, provider.name);
      return result;
    } catch (error) {
      if (params.signal?.aborted) throw params.signal.reason ?? error;
      if (error instanceof Error && error.message.includes("UNSUPPORTED_CONTENT")) throw error;
      // Never return SDK bodies, request headers, or chart prompts to the UI.
      const message = error instanceof Error ? error.message : "";
      const code = message.match(/\b[A-Z][A-Z_]+:/)?.[0] ?? "REQUEST_FAILED:";
      const status = message.match(/\bHTTP \d{3}\b/)?.[0] ?? (deadline.aborted ? "timeout" : "request failed");
      failures.push(`${provider.name} ${code} ${status}`);
    }
  }
  if (candidates.length) throw new Error(`AI_PROVIDERS_FAILED: ${failures.join("; ")}`);
  throw new Error("AI_FREE_PROVIDER_UNCONFIGURED: Configure GEMINI_API_KEY, GROQ_API_KEY, or OPENROUTER_API_KEY.");
}

async function ask(

function readerFacingIntelligence(raw: string) {
  const cleaned = raw
    .trim()
    .replace(/^```(?:json)?\s*/i, "")
    .replace(/\s*```$/, "");
  try {
    const data = JSON.parse(cleaned) as Record<string, unknown>;
    const section = (title: string, key: string) => {
      const values = Array.isArray(data[key])
        ? (data[key].filter(item => typeof item === "string") as string[])
        : [];
      return values.length
        ? `### ${title}\n${values.map(value => `- ${value}`).join("\n")}`
        : "";
    };
    return [
      "## The shape of your chart",
      "This is a concise map of the strongest patterns the calculation supports. The longer reading below turns these signals into a personal story.",
      section("What stands out", "dominantPlanets"),
      section("Life areas being emphasized", "dominantHouses"),
      section("Signs and qualities in focus", "dominantSigns"),
      section("The repeating thread", "repeatedThemes"),
      section("Strongest evidence", "strongestEvidence"),
      section("The central tensions", "tensions"),
      section("Placements to keep in view", "priorityPlacements"),
      section("What may be active now", "currentTransitThemes"),
      section("Psychological possibilities to test", "psychologicalHypotheses"),

  } catch {
    return raw
      .replace(/[{}\[\]\"]/g, "")
      .replace(/,\s*/g, "\n")
      .trim();
  }
}

// Maps a model-written evidence entry onto a real chart point, and rewrites it from the chart data so facts cannot drift.
function resolvePoint(entry: unknown, known: Map<string, string>) {
  if (typeof entry !== "string") return null;
  const text = entry.trim().toLowerCase();
  const named = known.get(text);
  if (named) return named;
  // Do not accept an invented aspect merely because it starts with "Moon".
  // Accept only a point name or the exact canonical fact printed on the sheet.
  return Array.from(known.values()).find(value => value.trim().toLowerCase() === text) ?? null;
}

export function evidenceAnchors(packet: AstrologyEvidencePacket, synthesis?: SynthesisResult) {
  const anchors = new Map<string, string>();
  const items = [
    ...packet.genesisAstroEvidence,
    ...packet.patternEvidence,
    ...packet.behavioralEvidence,
    ...packet.lifeEventEvidence,
    ...packet.westernEvidence,
    ...packet.vedicEvidence,
    ...packet.arabicEvidence,
    ...packet.timingEvidence,
    ...packet.fixedStarEvidence,
    ...packet.lunarEvidence,
    ...packet.canonEvidence,
    ...packet.uncertainties,
  ];
  const subjectCounts = new Map<string, number>();
  for (const item of items) {
    const subject = item.subject.trim().toLowerCase();
    subjectCounts.set(subject, (subjectCounts.get(subject) ?? 0) + 1);
  }
  for (const item of items) {
    const subject = item.subject.trim().toLowerCase();
    // A subject can name multiple independent testimonies: never silently pick one.
    if (subjectCounts.get(subject) === 1) anchors.set(subject, item.statement);
    anchors.set(item.id.trim().toLowerCase(), item.statement);
    anchors.set(item.statement.trim().toLowerCase(), item.statement);
  }
  // Include exactly the finding and Canon IDs that runInterpretation sends.
  // Repeated subject names remain ambiguous; exact IDs are always usable.
  const add = (id: string, statement: string) => {
    if (id.trim() && statement.trim()) anchors.set(id.trim().toLowerCase(), statement);
  };
  for (const method of synthesis?.methodOutcomes ?? []) {
    for (const finding of method.findingJudgments) {
      add(finding.id, finding.judgment);
      for (const id of finding.evidenceIds) {
        const calculated = method.calculatedFindings.find(item => item.evidenceId === id);
        add(id, calculated ? `${calculated.label}: ${calculated.calculatedValue}` : finding.judgment);
      }
    }
    for (const finding of method.calculatedFindings) add(finding.evidenceId, `${finding.label}: ${finding.calculatedValue}`);
  }
  for (const rule of synthesis?.canonApplications ?? []) add(rule.canonEvidenceId, rule.statement);
  return anchors;
}

function resolveSemanticEvidence(entry: string, semantic: Map<string, string>) {
  const text = entry.trim().toLowerCase();
  const exact = semantic.get(text);
  if (exact) return exact;
  const tokens = text
    .replace(/[^a-z0-9]+/g, " ")
    .split(" ")
    .filter(token => token && token !== "in" && token !== "the");
  if (!tokens.length) return null;
  for (const [key, value] of Array.from(semantic.entries())) {
    const keyTokens = key
      .replace(/[^a-z0-9]+/g, " ")
      .split(" ")
      .filter(token => token && token !== "in" && token !== "the");
    if (
      keyTokens.length === tokens.length &&
      keyTokens.every(token => tokens.includes(token))
    )
      return value;
  }
  return null;
}

type ReadingPlan = {
  threads: Array<{
    title: string;
    insight: string;
    evidence: string[];
    chapters: string[];
  }>;
  tensions: Array<{ between: string; insight: string; evidence: string[] }>;
  chapters: Record<string, { angle: string; evidence: string[] }>;
};

export function validatePlan(
  raw: string,
  known: Map<string, string>,
  semantic: Map<string, string>
): ReadingPlan | null {
  const start = raw.indexOf("{");
  const end = raw.lastIndexOf("}");
  if (start < 0 || end <= start) return null;
  let data: any;
  try {
    data = JSON.parse(raw.slice(start, end + 1));
  } catch {
    return null;
  }
  if (!data || typeof data !== "object" || Array.isArray(data)) return null;
  const chapterIds = new Set<string>(
    READING_CHAPTERS.map(chapter => chapter.id)
  );
  const clean = (list: unknown) => {
    const out = new Set<string>();
    for (const entry of Array.isArray(list) ? list : []) {
      const semanticHit = typeof entry === "string" ? resolveSemanticEvidence(entry, semantic) : null;
      const hit = semanticHit ?? resolvePoint(entry, known);
      if (hit) out.add(hit);
      // Unknown evidence is not a chart fact: never carry it into the validated plan.
    }
    return Array.from(out);
  };
  const text = (value: unknown) =>
    typeof value === "string" ? value.trim() : "";
  const chapterAliases: Record<string, string> = {
    "mind-heart": "family-roots",
    destiny: "turning-points",
    mirror: "blessings-lessons",
  };
  const resolveChapterId = (rawId: unknown): string | null => {
    if (typeof rawId !== "string") return null;
    if (chapterIds.has(rawId)) return rawId;
    const mapped = chapterAliases[rawId];
    return mapped && chapterIds.has(mapped) ? mapped : null;
  };
  const threads = (Array.isArray(data.threads) ? data.threads : [])
    .map((thread: any) => ({
      title: text(thread?.title),
      insight: text(thread?.insight),
      evidence: clean(thread?.evidence),
      chapters: (Array.isArray(thread?.chapters) ? thread.chapters : [])
        .map(resolveChapterId)
        .filter((id: string | null): id is string => Boolean(id)),
    }))
    .filter(
      (thread: { title: string; insight: string; evidence: string[]; chapters: string[] }) =>
        thread.title && thread.insight && thread.evidence.length && thread.chapters.length
    );
  const tensions = (Array.isArray(data.tensions) ? data.tensions : [])
    .map((tension: any) => ({
      between: text(tension?.between),
      insight: text(tension?.insight),
      evidence: clean(tension?.evidence),
    }))
    .filter(
      (tension: { between: string; insight: string; evidence: string[] }) =>
        tension.between && tension.insight && tension.evidence.length
    );
  const chapters: ReadingPlan["chapters"] = {};
  const rawChapters = data.chapters ?? {};
  for (const [rawId, item] of Object.entries(rawChapters)) {
    const id = resolveChapterId(rawId);
    if (!id || chapters[id]) continue;
    const evidence = clean((item as any)?.evidence);
    if (item && text((item as any).angle) && evidence.length) {
      chapters[id] = { angle: text((item as any).angle), evidence };
    }
  }
  if (threads.length < 2 || Object.keys(chapters).length < 4) return null;
  return { threads, tensions, chapters };
}

function chapterPlan(analysis: string, chapterId: string) {
  if (!analysis) return "";
  let plan: ReadingPlan;
  try {
    plan = JSON.parse(analysis) as ReadingPlan;
  } catch {
    return "";
  }
  const isSynthesis = chapterId === "blessings-lessons" || chapterId === "mirror";
  const threads = isSynthesis
    ? plan.threads
    : plan.threads.filter(thread => thread.chapters.includes(chapterId));
  const mine = plan.chapters[chapterId];
  const out = [
    "READING PLAN (decided after reading the whole chart; follow it)",
  ];
  if (threads.length)
    out.push(
      isSynthesis
        ? "All threads to weave together:"
        : "Threads this chapter carries:",
      ...threads.map(
        thread =>
          `- ${thread.title}: ${thread.insight} Evidence: ${thread.evidence.join("; ")}.`
      )
    );
  if (plan.tensions.length)
    out.push(
      "Tensions in the chart:",
      ...plan.tensions.map(
        tension =>
          `- ${tension.between}: ${tension.insight} Evidence: ${tension.evidence.join("; ")}.`
      )
    );
  if (mine && !isSynthesis)
    out.push(
      `Evidence that is the main subject of THIS chapter: ${mine.evidence.join("; ")}.`,
      `Angle for this chapter: ${mine.angle}`,
      "Other chapters own the remaining placements. Mention one of them here only briefly, to connect threads."
    );
  if (isSynthesis)
    out.push(
      "This synthesis chapter owns no single placement of its own. It names where life flows naturally, where the hardest lessons repeat, what the reader contributes to trouble, and honest practical direction."
    );

  question = "",
  synthesisContext = ""
) {

    "mind-heart": "family-roots",

    mirror: "blessings-lessons",
  };

  const masterContext = formatCompactMasterGateContext(master);

    mode
  );
  const facts = `${masterContext}\n\n${buildEvidenceSheet(chart, mode)}\n\n${formatCompactEvidencePacket(evidencePacket)}\n\n${synthesisContext}`;
  const condensedChapters = completedChapters.map((c, idx) =>
    idx < completedChapters.length - 2 && c.length > 1600
      ? c.slice(0, 1500) + "…"
      : c
  );
  const isSynthesisChapter = resolvedId === "blessings-lessons";
  const context = condensedChapters.length
    ? `Already written chapters, below. Do not repeat any point, placement meaning, scene, or advice already made in them. " This is the final synthesis chapter: connect the whole chart into your greatest blessings, hardest recurring lessons, and where to go from here.     : "This is the opening chapter; establish the real human portrait cleanly.";

  try {
    const text = await ask(
      master.synthesis,
      "READING_CHAPTER",
      [
        { role: "system", content: chapterSystem(mode, chapter) },
        {
          role: "user",
          content: `${facts}\n\n${chapterPlan(analysis, resolvedId)}\n\n${context}\n\nWrite the complete ${chapter.title} chapter now.`,

function compactText(text: string, maxLen = 165): string {
  const clean = text.replace(/\s+/g, " ").trim();
  if (clean.length <= maxLen) return clean;
  const slice = clean.slice(0, maxLen - 1);
  const stop = Math.max(slice.lastIndexOf(". "), slice.lastIndexOf("; "), slice.lastIndexOf(" "));
  return (stop > maxLen * 0.55 ? slice.slice(0, stop) : slice).trim() + "…";
}

export function formatCompactEvidencePacket(packet: AstrologyEvidencePacket): string {
  const section = (title: string, items: EvidenceItem[], maxItems = 8) => {
    if (!items.length) return "";
    const sorted = [...items].sort((a, b) => (b.strength ?? 0) - (a.strength ?? 0));
    const sliced = sorted.slice(0, maxItems);
    const lines = sliced.map(
      item => `- [${item.tradition}/${item.technique}] ${compactText(item.statement)}`
    );
    if (items.length > maxItems) {
      lines.push(`- (+${items.length - maxItems} additional calculated items in synthesis)`);
    }
    return `${title}\n${lines.join("\n")}`;
  };

  const relationLines = (
    title: string,
    items: AstrologyEvidencePacket["relationships"],
    maxItems = 5
  ) => {
    if (!items.length) return "";
    return `${title}\n${items
      .slice(0, maxItems)
      .map(item => `- ${compactText(item.explanation, 150)} (strength ${item.strength})`)
      .join("\n")}`;
  };

  return [
    "ASTROLOGY EVIDENCE PACKET (COMPACT DIGEST)",
    `Question: ${packet.question}`,
    `Plan: ${packet.plan.reasons.join(" ")}`,
    section("Western evidence", packet.westernEvidence, 8),
    section("Genesis Astro Engine evidence", packet.genesisAstroEvidence, 8),
    section("Genesis Pattern Engine evidence", packet.patternEvidence, 8),
    section("Human Behavior Pattern Intelligence", packet.behavioralEvidence, 8),
    section("Life-Event & Situation Intelligence", packet.lifeEventEvidence, 5),
    section("Vedic evidence", packet.vedicEvidence, 8),
    section("Arabic evidence", packet.arabicEvidence, 6),
    section("Lunar & Fixed-Frame Overlays", packet.lunarEvidence, 8),
    section("Timing evidence (transits <= 5°)", packet.timingEvidence, 10),
    relationLines("Convergences", packet.convergences, 5),
    relationLines("Contradictions and qualifications", packet.contradictions, 5),
    packet.uncertainties.length
      ? section("Explicit uncertainties", packet.uncertainties, 4)
      : "",
    packet.incomplete.length
      ? `Incomplete or reference-only: ${packet.incomplete.slice(0, 4).join(" | ")}`
      : "",
  ]
    .filter(Boolean)
    .join("\n\n");
}

function computeSnapshotId(input: MasterChartInput, version: string): string {
  const content = JSON.stringify({
    location: input.location,
    lat: input.latitude,
    lon: input.longitude,
    tz: input.timezone,
    d: input.date,
    t: input.time,
    tl: input.transitLocation ?? "",
    tlat: input.transitLatitude ?? null,
    tlon: input.transitLongitude ?? null,
    ttz: input.transitTimezone ?? "",
    td: input.transitDate ?? "",
    tt: input.transitTime ?? "",
    instant: input.transitInstantUtc ?? "",
    baseInstant: input.instantUtc ?? "",
    birthTimeKnown: input.birthTimeKnown, chartRole: input.chartRole, elevation: input.elevation,
    m: input.mode ?? "natal",
    w: input.worldview ?? "agent-vs-god",
    sunriseAnchor: input.sunriseAnchor ?? null,
    transitSunriseAnchor: input.transitSunriseAnchor ?? null,
    xyz: input.xyzPositions ? Object.entries(input.xyzPositions).sort(([a], [b]) => a.localeCompare(b)) : null,
    transitXYZ: input.transitXYZPositions ? Object.entries(input.transitXYZPositions).sort(([a], [b]) => a.localeCompare(b)) : null,
    role: input.chartRole ?? "natal",
    v: version,
  });
  return "umc-" + crypto.createHash("sha256").update(content).digest("hex").slice(0, 16);
}

// Reuse the same versioned natal foundation across reading requests in this
// server process. Event/transit snapshots are deliberately never cached here.
// A bounded cache prevents unbounded growth; persistence is a separate concern.
const NATAL_CACHE_LIMIT = 64;
const natalSnapshotCache = new Map<string, Promise<UnifiedMasterChartSnapshot>>();

export async function getMasterNatalChart(
  input: MasterChartInput
): Promise<UnifiedMasterChartSnapshot> {
  const natalInput: MasterChartInput = {
    ...input,
    mode: "natal",
    worldview: "agent-vs-god",
    transitDate: undefined,
    transitTime: undefined,
    transitInstantUtc: undefined,
    transitLocation: undefined,
    transitLatitude: undefined,
    transitLongitude: undefined,
    transitTimezone: undefined,
    transitSunriseAnchor: undefined,
    transitXYZPositions: undefined,
  };
  const key = computeSnapshotId(natalInput, "firmament-master-2026.3");
  const cached = natalSnapshotCache.get(key);
  if (cached) {
    natalSnapshotCache.delete(key);
    natalSnapshotCache.set(key, cached);
    return cached;
  }
  const pending = buildUnifiedMasterChart(natalInput);
  natalSnapshotCache.set(key, pending);
  while (natalSnapshotCache.size > NATAL_CACHE_LIMIT) {
    const oldest = natalSnapshotCache.keys().next().value;
    if (oldest === undefined) break;
    natalSnapshotCache.delete(oldest);
  }
  try {
    return await pending;
  } catch (error) {
    if (natalSnapshotCache.get(key) === pending) natalSnapshotCache.delete(key);
    throw error;
  }
}

import type { ChartResult, ChartRow } from "../astronomy";

/**
 * Single, read-only distribution contract for all Master Chart calculation layers.
 * The astronomy engine calculates once; layer adapters consume this shared context.
 * Never mutate the source chart or independently recalculate its positions here.
 */
export interface MasterChartDataBus {
  readonly chart: ChartResult;
  readonly planets: readonly ChartRow[];
  readonly houses: readonly number[];
  readonly ascendant: ChartRow | null;
  readonly sun: ChartRow;
  readonly moon: ChartRow;
  readonly transits: readonly ChartRow[];
}

/** Fail explicitly instead of substituting a fabricated angle or body position. */
export function createMasterChartDataBus(chart: ChartResult): MasterChartDataBus {
  const ascendant = chart.ascendant;
  if (chart.worldview !== "god" && (!ascendant || !Number.isFinite(ascendant.longitude))) {
    throw new Error("MASTER_CHART_MISSING_ASCENDANT");
  }
  if (chart.houses.length !== 12 || chart.houses.some(h => !Number.isFinite(h))) {
    throw new Error("MASTER_CHART_INVALID_HOUSES");
  }
  const sun = chart.movingBodies.find(p => p.name === "Sun");
  const moon = chart.movingBodies.find(p => p.name === "Moon");
  if (!sun || !Number.isFinite(sun.longitude)) throw new Error("MASTER_CHART_MISSING_SUN");
  if (!moon || !Number.isFinite(moon.longitude)) throw new Error("MASTER_CHART_MISSING_MOON");
  return Object.freeze({
    chart,
    planets: chart.movingBodies,
    houses: chart.houses,
    ascendant,
    sun,
    moon,
    transits: chart.transits ?? [],
  });
}

export function registeredMethodLedger(
  layers: Record<string, LayerRecord>
): LayerRecord[] {
  const missing = MASTER_METHOD_IDS.filter(id => !layers[id]);

    id => !(MASTER_METHOD_IDS as readonly string[]).includes(id)

    );

}

// A missing sunrise (polar day/night) is an error, never a substituted angle.

export function noaaSunriseUtc(localDate: string, latitude: number, longitude: number, setting = false): Date {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(localDate) || !Number.isFinite(latitude) || Math.abs(latitude) > 90 || !Number.isFinite(longitude) || Math.abs(longitude) > 180)

    const t = (jd - 2451545) / 36525;
    const l0 = ((280.46646 + t * (36000.76983 + t * 0.0003032)) % 360 + 360) % 360;
    const m = (357.52911 + t * (35999.05029 - 0.0001537 * t)) * rad;
    const ecc = 0.016708634 - t * (0.000042037 + 0.0000001267 * t);

    const omega = (125.04 - 1934.136 * t) * rad;
    const lambda = (trueLong - 0.00569 - 0.00478 * Math.sin(omega)) * rad;
    const obliq0 = 23 + (26 + (21.448 - t * (46.815 + t * (0.00059 - t * 0.001813))) / 60) / 60;
    const obliq = (obliq0 + 0.00256 * Math.cos(omega)) * rad;
    const decl = Math.asin(Math.sin(obliq) * Math.sin(lambda));
    const y = Math.tan(obliq / 2) ** 2;
    const l = l0 * rad;
    const eqtime = 4 * (180 / Math.PI) * (y * Math.sin(2 * l) - 2 * ecc * Math.sin(m) + 4 * ecc * y * Math.sin(m) * Math.cos(2 * l) - 0.5 * y * y * Math.sin(4 * l) - 1.25 * ecc * ecc * Math.sin(2 * m));
    const cosH = (Math.cos(90.833 * rad) - Math.sin(latitude * rad) * Math.sin(decl)) / (Math.cos(latitude * rad) * Math.cos(decl));
    if (!Number.isFinite(cosH) || cosH < -1 || cosH > 1) throw new UndefinedSolarEvent("NOAA sunrise undefined for this location/date; no fallback.");
    return 720 - 4 * (longitude + (setting ? -1 : 1) * Math.acos(cosH) / rad) - eqtime;

    }
  }
  if (lastError instanceof Error && lastError.message === "No matching place found.") throw lastError;

import type { EvidenceItem, EvidenceRelationship, Tradition } from "./types";

const traditions = ["western", "vedic", "arabic"] as const;
function key(evidence: EvidenceItem) {
  const subject = evidence.subject.toLowerCase();
  const structuralSubject = [
    "sun",
    "moon",
    "venus",
    "mars",
    "jupiter",
    "saturn",
    "ascendant",
    "relationship",
    "career",
    "fortune",
    "spirit",
    "timing",
  ].find(value => subject.includes(value));
  if (structuralSubject) return structuralSubject;
  return (
    evidence.concepts
      .filter(value => !["neutral", "incomplete"].includes(value))
      .find(value =>
        /house-|moon|sun|venus|mars|jupiter|saturn|relationship|career|lunar|dignity|condition|aspect|drishti|lot|manzil/.test(
          value
        )
      ) ?? evidence.technique
  );
}
function relationFor(items: EvidenceItem[]): EvidenceRelationship | null {
  const present = Array.from(
    new Set(
      items
        .map(item => item.tradition)
        .filter((tradition): tradition is Tradition =>
          traditions.includes(tradition as (typeof traditions)[number])
        )
    )
  );
  if (present.length < 2) return null;
  const pressured = items.some(item => item.polarity === "pressured");
  const supportive = items.some(item => item.polarity === "supportive");
  const relation =
    pressured && supportive
      ? "qualification"
      : pressured
        ? "reinforcement"
        : "reinforcement";
  const theme = key(items[0]);
  return {
    id: `relationship-${theme}-${present.join("-")}`,
    relation,
    theme,
    evidenceIds: items.map(item => item.id),
    traditions: present,
    explanation:
      relation === "qualification"
        ? `The traditions describe ${theme} through different layers: some evidence supports capacity while other evidence adds pressure or work.`
        : `Independent evidence in ${present.join(", ")} describes the same structural subject, so the relationship is stronger than a shared keyword alone.`,
    strength: Number(
      Math.min(
        1,
        0.35 +
          present.length * 0.2 +
          (items.reduce((sum, item) => sum + (item.strength ?? 0), 0) /
            Math.max(1, items.length)) *
            0.3
      ).toFixed(3)
    ),
  };
}

      polarities.has("pressured") &&

    relationships,

  };

export type Varga = "D2" | "D3" | "D9" | "D10";

export function vargaPlacement(longitude: number, division: Varga) {
  const lon = normalizeLongitude(longitude), sign = Math.floor(lon / 30), degree = lon % 30;
  const n = Number(division.slice(1)), part = Math.min(n - 1, Math.floor(degree / (30 / n)));
  let result: number;
  if (division === "D2") {
    const sunHora = (sign % 2 === 0) === (part === 0);
    result = sunHora ? 4 : 3;
  } else if (division === "D3") {
    result = (sign + 4 * part) % 12;
  } else if (division === "D9") {
    const start = sign % 3 === 0 ? sign : sign % 3 === 1 ? (sign + 8) % 12 : (sign + 4) % 12;
    result = (start + part) % 12;
  } else {
    result = ((sign % 2 === 0 ? sign : (sign + 8) % 12) + part) % 12;
  }
  return {
    division,
    part: part + 1,
    sourceSign: ZODIAC_SIGNS[sign],
    sign: ZODIAC_SIGNS[result],
    signIndex: result,
    degreeInPart: degree - part * (30 / n),
    coordinateFrame: "guardian-fixed-zero-reference" as const,
  };
}

export function grahaDrishtiTargetHouses(planet: string, fromHouse: number): number[] {

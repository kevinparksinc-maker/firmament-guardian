/** All ordinary chart readings enter through the same calculated method ledger. */
async function synthesizeChartForReadingUncached(
  chart: ChartResult,
  mode: "natal" | "transit" | "combined",
  question: string
): Promise<{
  snapshot: UnifiedMasterChartSnapshot;
  synthesis: SynthesisResult;
  context: string;
  canonicalChart: ChartResult;
}> {
  const input = {
    ...chart.input,
    worldview: chart.worldview,
    mode,
    instantUtc: chart.utc,
    ...(mode === "natal" ? {} : { transitInstantUtc: chart.transitDate }),
  };
  if (
    !input.date ||
    (chart.worldview !== "god" &&
      (!input.location ||
        !input.time ||
        !input.timezone ||
        !Number.isFinite(input.latitude) ||
        !Number.isFinite(input.longitude)))
  ) {
    throw new Error(
      "MASTER_INPUT_MISSING: Resolved chart date, time, coordinates and timezone are required; no substitute data is used."
    );
  }
  const snapshot =
    mode === "natal" && chart.worldview === "agent-vs-god"
      ? await getMasterNatalChart(input)
      : await buildUnifiedMasterChart(input);
  // Do not silently interpret a second chart if an earlier client chart differs.
  for (const body of chart.movingBodies) {
    const canonical = snapshot.baseCelestialState.planets.find(
      row => row.name === body.name
    );
    if (!canonical)
      throw new Error(
        `Master Chart is missing ${body.name}; cannot synthesize a divergent reading.`
      );
    const difference = Math.abs(
      ((canonical.longitude - body.longitude + 540) % 360) - 180
    );
    if (difference > 0.01)
      throw new Error(
        `Master Chart mismatch for ${body.name}: ${difference.toFixed(4)}°. Refresh the chart before reading.`
      );
  }
  const angularDifference = (a: number, b: number) =>
    Math.abs(((a - b + 540) % 360) - 180);
  if (
    chart.ascendant &&
    (!snapshot.baseCelestialState.ascendant ||
      angularDifference(
        snapshot.baseCelestialState.ascendant.longitude,
        chart.ascendant.longitude
      ) > 0.01)
  ) {
    throw new Error(
      "Master Chart Ascendant mismatch. Refresh the chart before reading."
    );
  }
  if (
    chart.houses.length !== (snapshot.natal?.houses ?? []).length ||
    chart.houses.some(
      (cusp, index) =>
        angularDifference(cusp, snapshot.natal!.houses[index]) > 0.01
    )
  ) {
    throw new Error(
      "Master Chart equal-house cusp mismatch. Refresh the chart before reading."
    );
  }
  if (
    mode !== "natal" &&
    chart.transits.length > 0 !== Boolean(snapshot.transit?.planets.length)
  ) {
    throw new Error(
      "Master Chart transit coverage mismatch. Refresh the chart before reading."
    );
  }
  if (mode !== "natal" && chart.transits.length) {
    for (const transit of chart.transits) {
      const canonical = snapshot.transit?.planets.find(
        row => row.name === transit.name
      );
      if (
        !canonical ||
        angularDifference(canonical.longitude, transit.longitude) > 0.01
      ) {
        throw new Error(
          `Master Chart transit mismatch for ${transit.name}. Refresh the chart before reading.`
        );
      }
    }
  }
  return synthesizeMasterSnapshot(snapshot, mode, question);
}

/** Assemble reading context from a server-created snapshot without recalculating it. */
export async function synthesizeMasterSnapshot(
  snapshot: UnifiedMasterChartSnapshot,
  mode: "natal" | "transit" | "combined",
  question: string
) {
  const result = await synthesizeSnapshotEvidence(snapshot, mode, question);
  return { ...result, context: formatSynthesisContext(result.synthesis) };
}

// Reuse concurrent/repeated reads of the exact same source chart and question.
// Client-supplied method bindings are never trusted as calculation evidence.
const verifiedReadings = new Map<
  string,
  {
    expiresAt: number;
    promise: ReturnType<typeof synthesizeChartForReadingUncached>;
  }
>();
export async function synthesizeChartForReading(
  chart: ChartResult,
  mode: "natal" | "transit" | "combined",
  question: string
) {
  const { methodBindings: _ignored, ...source } = chart;
  const key = createHash("sha256")
    .update(JSON.stringify({ source, mode, question }))
    .digest("hex");
  const cached = verifiedReadings.get(key);
  if (cached && cached.expiresAt > Date.now()) return cached.promise;
  const promise = synthesizeChartForReadingUncached(chart, mode, question);
  verifiedReadings.set(key, { expiresAt: Date.now() + 60_000, promise });
  while (verifiedReadings.size > 8)
    verifiedReadings.delete(verifiedReadings.keys().next().value!);
  try {
    return await promise;
  } catch (error) {
    if (verifiedReadings.get(key)?.promise === promise)
      verifiedReadings.delete(key);
    throw error;
  }
}

import type { UnifiedMasterChartSnapshot } from "../../masterChart/types";
import {
  canonDossierToEvidenceItems,
  retrieveChartCanonDossier,
} from "../canon/retriever";
import { buildCanonRuleApplications } from "./canonApplications";

/** Server-created snapshots enter here directly: no second ephemeris pass. */
export async function synthesizeSnapshotEvidence(
  snapshot: UnifiedMasterChartSnapshot,
  mode: "natal" | "transit" | "combined",
  question: string
) {
  // The Master Chart remains the only calculation source. Its canonical natal
  // ChartResult is passed to the existing deterministic retriever; no second
  // astronomical calculation or alternate house/frame pipeline is introduced.
  const canonicalChart = snapshot.natal;
  if (!canonicalChart)
    throw new Error("MASTER_CHART_MISSING: Snapshot has no canonical chart.");
  const canonDossier = retrieveChartCanonDossier(canonicalChart);
  const canonEvidence = canonDossierToEvidenceItems(canonDossier);
  const canonApplications = buildCanonRuleApplications(snapshot, canonEvidence);
  // The same registry is evaluated for each sky. Namespace event evidence so
  // Saturn's present mansion can never be mistaken for natal Saturn's mansion.
  const event = snapshot.transit?.framework;
  const eventLayers =
    event?.methodExecutionLedger.map(layer => ({
      ...layer,
      id: `transit:${layer.id}`,
      name: `Current sky · ${layer.name}`,
      frame: layer.frame,
      evidenceIds: layer.evidenceIds.map(id => `transit:${id}`),
      prerequisites: layer.prerequisites.map(id => `transit:${id}`),
    })) ?? [];
  if (event?.natal) {
    const applications = buildCanonRuleApplications(
      event,
      canonDossierToEvidenceItems(retrieveChartCanonDossier(event.natal))
    );
    canonApplications.push(
      ...applications.map(application => ({
        ...application,
        canonEvidenceId: `transit:${application.canonEvidenceId}`,
        subject: `Current sky ${application.subject}`,
        calculationEvidenceIds: application.calculationEvidenceIds.map(
          id => `transit:${id}`
        ),
      }))
    );
  }
  const synthesisSnapshot = {
    ...snapshot,
    layers: {
      ...snapshot.layers,
      ...Object.fromEntries(eventLayers.map(layer => [layer.id, layer])),
    },
    methodExecutionLedger: [...snapshot.methodExecutionLedger, ...eventLayers],
  };
  const synthesis = await synthesizeMasterReading({
    snapshot: synthesisSnapshot,
    readingMode: mode,
    userQuestion: question,
    canonApplications,
  });
  if (!synthesis.qualityAudit.passed) {
    console.warn(
      "[SYNTHESIS_THEME_COVERAGE] Rule-based theme coverage is incomplete; forwarding all calculated results to the AI:",
      synthesis.qualityAudit.unsupportedClaims
    );
  }
  return { snapshot, synthesis, canonicalChart };
}

import type {
  SynthesisInput,
  SynthesisResult,
} from "./types";
import { extractPropositionsFromMethodOutcomes } from "./normalizer";
import { clusterPropositionsIntoThemes } from "./clustering";
import { buildMethodOutcomes } from "./methodOutcomes";

export async function synthesizeMasterReading(
  input: SynthesisInput
): Promise<SynthesisResult> {
  const { snapshot, readingMode } = input;

  // 1. Extract normalized propositions across all available traditions and methods
  const suppliedCanon = input.canonApplications ?? [];
  const methodOutcomes = buildMethodOutcomes(snapshot, suppliedCanon);
  const canonApplications = Array.from(
    new Map(
      [...suppliedCanon, ...methodOutcomes.flatMap(outcome => outcome.canonApplications)].map(
        rule => [rule.canonEvidenceId, rule]
      )
    ).values()
  );
  const propositions = extractPropositionsFromMethodOutcomes(methodOutcomes);

  // 2. Cluster into human domain themes and reconcile tensions
  const { themes, tensions } = clusterPropositionsIntoThemes(propositions);

  // 3. Evidence-led synthesis. Never substitute generic personality prose for findings.
  const question = input.userQuestion?.trim() || "the chart's overall configuration";
  const relevantDomains: Record<string, string[]> = {
    relationships: ["relationship", "marriage", "partner", "love"],
    resources_wealth: ["money", "wealth", "finance", "income", "career pay"],
    purpose_work: ["career", "work", "purpose", "job", "business"],
    family_home: ["family", "home", "parent", "child"],
    timing: ["when", "timing", "today", "transit", "future"],
    energy_wellbeing: ["energy", "health", "wellbeing"],
    autonomy_power: ["power", "conflict", "control", "courage"],
  };
  const query = question.toLowerCase();
  const prioritized = Object.entries(relevantDomains)
    .filter(([, terms]) => terms.some(term => query.includes(term)))
    .map(([domain]) => domain);
  const strengthWeight = { dominant: 4, strong: 3, moderate: 2, subtle: 1 };
  const signal = (theme: (typeof themes)[number]) =>
    Math.max(
      0,
      ...propositions
        .filter(p => theme.linkedPropositionIds.includes(p.id))
        .map(p => strengthWeight[p.strength])
    );
  const orderedThemes = [...themes].sort(
    (a, b) =>
      (prioritized.includes(b.domain) ? 1 : 0) - (prioritized.includes(a.domain) ? 1 : 0) ||
      signal(b) - signal(a)
  );
  const methodFindings = snapshot.methodExecutionLedger.map(layer => {
    const raw = layer.results;
    const entries = Array.isArray(raw)
      ? raw.map((value, index) => [String(index + 1), value] as const)
      : raw && typeof raw === "object"
        ? Object.entries(raw)
        : raw == null
          ? []
          : [["result", raw] as const];
    const findings = entries.map(([label, value]) => ({
      label,
      details: typeof value === "string" ? value : JSON.stringify(value) ?? String(value),
    }));
    return {
      id: layer.id,
      name: layer.name,
      tradition: layer.tradition,
      status: layer.status,
      frame: layer.frame,
      evidenceIds: layer.evidenceIds,
      sourceIds: layer.sourceIds,
      resultCount: findings.length,
      findings,
      ...(layer.notes ? { notes: layer.notes } : {}),
    };
  });
  const methodCounts = snapshot.methodExecutionLedger.map(
    layer =>
      `${layer.name}: ${layer.status}${Array.isArray(layer.results) ? ` (${layer.results.length} results)` : ""}`
  );
  const isInternal = !question || /^(Compact|the chart's|Placement summaries|Synastry:)/i.test(question);
  const activeMethodCount = methodOutcomes.filter(
    outcome => outcome.status === "interpreted" || outcome.findingJudgments.length > 0
  ).length;
  const dominantConclusion = orderedThemes.length
    ? isInternal
      ? `The celestial configuration is most clearly expressed through ${orderedThemes[0].humanTheme}. ${orderedThemes
          .slice(0, 3)
          .map(theme => theme.primaryConclusion)
          .join(" ")}`
      : `Regarding your inquiry ("${question}"): The celestial configuration is most clearly expressed through ${orderedThemes[0].humanTheme}. ${orderedThemes
          .slice(0, 3)
          .map(theme => theme.primaryConclusion)
          .join(" ")}`
    : activeMethodCount > 0
      ? `Unclustered method outcomes are active across ${activeMethodCount} layer(s), though 0 cross-domain themes were clustered.`
      : `The celestial configuration currently shows no qualifying interpreted method outcomes.`;

  // The complete recorded results remain in the propositions and snapshot.
  // Emit each computed finding so no doctrine disappears behind a generic summary.
  const integratedNarrative: string[] = [
    `Method inventory (every recorded layer): ${methodCounts.join("; ")}.`,
    ...methodOutcomes.map(
      outcome =>
        `METHOD OUTCOME [${outcome.methodId}] scope=${outcome.chartScope} frame=${outcome.frame} status=${outcome.status}. Interpretation: ${outcome.interpretation} Judgment: ${outcome.judgment} Contribution: ${outcome.contribution} Evidence: ${outcome.evidenceIds.join(", ") || "none"}. Limitations: ${outcome.strengthAndLimitations.join(" | ") || "none"}.`
    ),
    ...canonApplications.map(
      application =>
        `Canon rule applied to ${application.subject}: [${application.ruleId}] ${application.statement} Source: ${application.sourceTitle}. Matched calculation evidence: ${application.calculationEvidenceIds.join(", ")}.`
    ),
    ...orderedThemes.map(
      theme =>
        `${theme.humanTheme}. WHAT: ${theme.primaryConclusion} WHY/HOW: ${theme.whyHow} CONSEQUENCE: ${theme.consequences} MEANING: ${theme.deeperMeaning} Evidence: ${theme.evidenceIds.join(", ")}.`
    ),

      .filter(p => p.id.startsWith("prop-method-"))
      .map(
        p =>
          `[${p.tradition} / ${p.causalMechanism}] ${p.plainLanguageMeaning} [Evidence: ${p.supportingEvidenceIds.join(", ")}]`
      ),
  ];

  // 5. Build Evidence Trace Map (mapping every theme to its underlying evidence IDs)
  const evidenceTraceMap: Record<string, string[]> = {};
  for (const theme of themes) {
    evidenceTraceMap[theme.id] = theme.evidenceIds;
  }
  evidenceTraceMap["dominant-conclusion"] = orderedThemes[0]?.evidenceIds ?? [];

  // 6. Summarize Method Ledger
  const ledger = snapshot.methodExecutionLedger;
  const methodLedgerSummary = {
    totalEvaluated: ledger.length,
    computedWithResult: ledger.filter(l => l.status === "computed-with-result").length,
    referenceOnly: ledger.filter(l => l.status === "reference-only").length,
    notApplicable: ledger.filter(l => l.status === "not-applicable").length,
    failed: ledger.filter(l => l.status === "failed").length,
  };

  // 7. Evidence coverage audit. Do not claim a perfect score without checking.
  const expectedEvidenceIds = new Set(
    ledger.filter(l => l.status === "computed-with-result").flatMap(l => l.evidenceIds)
  );
  // Only evidence actually incorporated into a synthesized theme counts as interpreted.
  // Inventory-only ledger entries cannot inflate the interpretation coverage score.
  const tracedEvidenceIds = new Set(themes.flatMap(theme => theme.evidenceIds));
  const missingEvidenceIds = Array.from(expectedEvidenceIds).filter(
    id => !tracedEvidenceIds.has(id)
  );
  const computedLayers = ledger.filter(l => l.status === "computed-with-result");
  const interpretedMethodIds = new Set(
    methodOutcomes
      .filter(
        outcome =>
          outcome.status === "interpreted" &&
          outcome.findingJudgments.every(finding =>
            themes.some(theme => theme.linkedPropositionIds.includes(finding.id))
          )
      )
      .map(outcome => outcome.methodId)
  );
  const uninterpretedMethods = computedLayers.filter(
    layer => !interpretedMethodIds.has(layer.id)
  );
  const linkedPropositionIds = new Set(themes.flatMap(theme => theme.linkedPropositionIds));
  const interpretedCanonEvidenceIds = new Set(
    propositions
      .filter(proposition => linkedPropositionIds.has(proposition.id))
      .flatMap(proposition =>
        (proposition.canonApplications ?? []).map(application => application.canonEvidenceId)
      )
  );
  const missingCanonApplications = canonApplications.filter(
    application =>
      application.confidence !== "reference-only" &&
      !interpretedCanonEvidenceIds.has(application.canonEvidenceId)
  );
  const incompleteMethodOutcomes = methodOutcomes.filter(outcome => outcome.status === "failed");
  const groundedRatio =
    expectedEvidenceIds.size === 0
      ? 1
      : (expectedEvidenceIds.size - missingEvidenceIds.length) / expectedEvidenceIds.size;
  const qualityAudit = {
    groundedRatio,
    unsupportedClaims: [
      ...incompleteMethodOutcomes.map(
        outcome => `Method judgment failed: ${outcome.methodId}: ${outcome.interpretation}`
      ),
      ...missingEvidenceIds.map(id => `Not incorporated into interpreted themes: ${id}`),
      ...uninterpretedMethods.map(
        layer => `Computed method missing from interpreted themes: ${layer.id}`
      ),
      ...missingCanonApplications.map(
        application =>
          `Canon rule not applied to an interpreted proposition: ${application.ruleId} (${application.canonEvidenceId})`
      ),
    ],
    passed:
      missingEvidenceIds.length === 0 &&
      uninterpretedMethods.length === 0 &&
      missingCanonApplications.length === 0 &&
      incompleteMethodOutcomes.length === 0,
  };
  // Report method coverage explicitly; reference-only or unavailable layers are never
  // described as calculated. The raw evidence remains available in the snapshot.
  integratedNarrative.push(
    `Calculation coverage: ${ledger.filter(l => l.status === "computed-with-result").length} methods returned results, ${methodLedgerSummary.referenceOnly} are reference-only, ${methodLedgerSummary.notApplicable} are not applicable, and ${methodLedgerSummary.failed} failed. ${missingEvidenceIds.length === 0 ? "All recorded computed evidence IDs are incorporated into interpreted themes." : `${missingEvidenceIds.length} computed evidence IDs are not incorporated into interpreted themes.`}`
  );
  integratedNarrative.push(
    missingCanonApplications.length === 0
      ? `Canon coverage: ${canonApplications.length} retrieved rule applications are linked to interpreted chart propositions.`
      : `Canon coverage incomplete: ${missingCanonApplications.length} retrieved rule applications are not linked to interpreted chart propositions.`
  );

  return {
    sourceSnapshot: snapshot,
    snapshotId: snapshot.snapshotId,
    mode: readingMode,
    dominantConclusion,
    integratedNarrative,
    methodFindings,
    methodOutcomes,
    methodOutcomeAudit: {
      complete: incompleteMethodOutcomes.length === 0,
      incompleteMethodIds: incompleteMethodOutcomes.map(outcome => outcome.methodId),
    },
    themes,
    contradictionsAndTensions: tensions,
    timingOverview:
      readingMode === "transit" || readingMode === "combined"
        ? {
            activeWindow: snapshot.transit?.transitInstantUtc ?? "Transit moment not supplied",
            turningPoints: snapshot.transit?.aspectsToNatal?.length
              ? snapshot.transit.aspectsToNatal
                  .map(
                    contact =>
                      `${contact.transitBody} ${contact.aspect} natal ${contact.natalBody} (orb ${Number(contact.orb).toFixed(2)}°)`
                  )
                  .slice(0, 24)
              : [
                  "No calculated natal transit contacts available; peak and easing times are not established.",
                ],
          }
        : undefined,
    methodLedgerSummary,
    evidenceTraceMap,
    canonApplications,
    qualityAudit,
  };
}

import type {

  UnifiedMasterChartSnapshot,
} from "../../masterChart/types";
import type {
  CanonRuleApplication,
  MethodOutcome,
  MethodOutcomeFinding,
} from "./types";
import { applyMethodRules } from "./methodRules";

const frameOf = (layer: LayerRecord): MethodOutcome["frame"] =>
  ["god", "agent", "transit"].includes(layer.frame)
    ? (layer.frame as MethodOutcome["frame"])
    : "shared";
function findings(layer: LayerRecord): MethodOutcomeFinding[] {
  const values = Array.isArray(layer.results)
    ? layer.results
    : layer.results == null
      ? []
      : [layer.results];
  return values.map((value, index) => ({
    evidenceId: layer.evidenceIds[index] ?? layer.evidenceIds[0] ?? layer.id,
    label: `${layer.name} finding ${index + 1}`,
    calculatedValue: typeof value === "string" ? value : JSON.stringify(value),
  }));
}
function outcome(
  layer: LayerRecord,
  snapshot: UnifiedMasterChartSnapshot,
  canon: CanonRuleApplication[]
): MethodOutcome {
  const event =
    layer.id.startsWith("transit:") ||
    layer.id === "transit-activation-layer" ||
    snapshot.inputs.chartRole === "event";
  const base = {
    methodId: layer.id,
    methodName: layer.name,
    tradition: layer.tradition,
    chartScope: event ? ("event" as const) : ("natal" as const),
    frame: frameOf(layer),
    calculatedFindings: findings(layer),
    findingJudgments: [],
    canonApplications: [],
    evidenceIds: layer.evidenceIds,
    strengthAndLimitations: layer.notes ? [layer.notes] : [],
    missingCanon: false,
  };
  if (layer.status === "reference-only")
    return {
      ...base,
      status: "reference-only",
      interpretation:
        layer.notes ??
        "Reference catalog; no executable personal placement method.",
      judgment:
        "No personal method judgment is issued from this reference-only catalog.",
      contribution: "Historical reference; excluded from interpreted themes.",
    };
  if (layer.status === "not-applicable")
    return {
      ...base,
      status: "not-applicable",
      interpretation:
        layer.notes ?? "This method's required chart scope is absent.",
      judgment: "No judgment for this scope.",
      contribution: "Not applicable; excluded from interpreted themes.",
    };
  if (layer.status === "computed-no-result")
    return {
      ...base,
      status: "no-result",
      interpretation:
        "The calculation completed and found no qualifying result.",
      judgment:
        "No positive activation was found under the method's configured conditions.",
      contribution:
        "Successful empty calculation; no personal indication inferred.",
    };
  if (layer.status !== "computed-with-result")
    return {
      ...base,
      status: "failed",
      interpretation:
        layer.notes ?? layer.errorCode ?? "The calculation is unavailable.",
      judgment: "Judgment blocked by the calculation failure.",
      contribution: "Calculation status is retained in the complete raw snapshot for AI interpretation.",
    };
  try {
    const judgments = applyMethodRules(layer, snapshot);
    if (!judgments.length)
      throw new Error(
        "METHOD_RULE_MISSING: No applicable finding judgments were produced."
      );
    const allIds = new Set(layer.evidenceIds);
    const linked = canon.filter(rule =>
      rule.calculationEvidenceIds.some(id => allIds.has(id))
    );
    // Link retrieved chart-specific doctrine to the outcome's actual judgments.
    // Detached rules remain unlinked and fail the final audit.
    for (const judgment of judgments)
      judgment.rules.push(
        ...linked.filter(rule =>
          rule.calculationEvidenceIds.some(id =>
            judgment.evidenceIds.includes(id)
          )
        )
      );
    // These producers already calculate relational/condition testimony from
    // explicit chart relationships. Admit that testimony as a judged modifier,
    // once in its owning method, rather than re-normalizing the raw snapshot.
    const owner: Record<string, string> = {
      "firmament-traditional-mechanisms": "western-planets-and-signs",
      "firmament-relational-chain": "western-houses-and-angles",
      "firmament-vedic-yogas": "genesis-behavioral-patterns",
      "firmament-arabic-persian-tradition": "arabic-hermetic-lots",
    };
    const localId = layer.id.replace(/^transit:/, "");
    const scopeIds = new Set(
      snapshot.methodExecutionLedger
        .filter(
          m => m.id.startsWith("transit:") === layer.id.startsWith("transit:")
        )
        .flatMap(m => m.evidenceIds)
    );
    for (const rule of canon.filter(
      rule =>
        owner[rule.ruleId] === localId &&
        rule.calculationEvidenceIds.some(id => scopeIds.has(id)) &&
        rule.implementationStatus !== "incomplete" &&
        rule.confidence !== "reference-only"
    )) {
      const ids = rule.calculationEvidenceIds.filter(id => scopeIds.has(id));
      const supportive = [
        "mutual-reception",
        "unilateral-reception",
        "translation-of-light",
        "collection-of-light",
        "cazimi",
      ].includes(rule.technique);
      const challenging = [
        "combustion",
        "besiegement",
        "prohibition",
        "frustration",
        "void-of-course",
      ].includes(rule.technique);
      const techniqueLabels: Record<string, string> = {
        "mutual-reception": "two placements actively supporting each other",
        "unilateral-reception": "one placement hosting and backing another",
        "translation-of-light": "a faster mover carrying momentum between two placements",
        "collection-of-light": "a shared planet bringing two separate placements into agreement",
        "cazimi": "close solar alignment sharpening focus",
        "combustion": "solar intensity overshadowing and internalizing expression",
        "besiegement": "pressure closing in from both sides",
        "prohibition": "an intervening influence interrupting direct follow-through",
        "frustration": "effort shifting direction before completion",
        "void-of-course": "a quiet pause before the next decisive push",
      };
      judgments.push({
        id: `prop-method-${layer.id}-canon-${rule.canonEvidenceId}`,
        subject: rule.subject,
        domain: judgments[0].domain,
        judgment: rule.statement,
        mechanism: techniqueLabels[rule.technique] ?? rule.technique,
        valence: supportive
          ? "supportive"
          : challenging
            ? "challenging"
            : "neutral",
        strength: "moderate",
        evidenceIds: ids,
        rules: [rule],
      });
    }
    const applications = Array.from(
      new Map(
        judgments
          .flatMap(j => j.rules)
          .map(rule => [rule.canonEvidenceId, rule])
      ).values()
    );
    const missingIds = layer.evidenceIds.filter(
      id => !judgments.some(j => j.evidenceIds.includes(id))
    );
    if (missingIds.length)
      throw new Error(
        `METHOD_EVIDENCE_UNINTERPRETED: ${missingIds.join(", ")}`
      );
    return {
      ...base,
      status: "interpreted",
      findingJudgments: judgments,
      canonApplications: applications,
      interpretation: judgments.map(j => j.judgment).join("\n\n"),
      judgment: judgments.map(j => j.judgment).join("\n\n"),
      contribution: `${judgments.length} finding-specific judgments contribute to ${Array.from(new Set(judgments.map(j => j.domain))).join(", ")}.`,
      strengthAndLimitations: [
        ...base.strengthAndLimitations,
        ...judgments.flatMap(j => (j.limitation ? [j.limitation] : [])),
      ],
    };
  } catch (error) {
    const reason = error instanceof Error ? error.message : String(error);
    return {
      ...base,
      status: "failed",
      missingCanon: reason.includes("RULE_MISSING") || reason.includes("CANON"),
      interpretation: reason,
      judgment:
        "No traditional judgment issued: this method failed rule application.",
      contribution:
        "Rule interpretation failed; calculated values remain available in the raw snapshot for AI interpretation.",
      strengthAndLimitations: [...base.strengthAndLimitations, reason],
    };
  }
}
export function buildMethodOutcomes(
  snapshot: UnifiedMasterChartSnapshot,
  canonApplications: CanonRuleApplication[]
): MethodOutcome[] {
  return snapshot.methodExecutionLedger.map(layer =>
    outcome(layer, snapshot, canonApplications)
  );

        if (!output.length)

            i,

            "No configured dominant pattern or occupied-house emphasis was returned. No personality or event judgment is inferred from an empty detection result.",

            "Local Genesis pattern detection",
            "choices_obstacles",
            "neutral",
            "subtle"
          );
      });
      break;
    default:
      throw new Error(`METHOD_RULE_MISSING: ${baseId}`);
  }
  return output;

import type { HumanDomain, InterpretedProposition, SynthesizedTheme } from "./types";

/**
 * Evidence-first synthesis. Every computed proposition is retained, while
 * meaningful interactions receive explicit support, tension, and weighting.
 * These are traditional chart interpretations, not assertions of life events.
 */
const DOMAIN_TITLES: Record<HumanDomain, string> = {
  identity: "Identity, Temperament & Self-Knowledge",
  purpose_work: "Purpose, Vocation & Gifts",
  relationships: "Relationships & Alliances",
  resources_wealth: "Resources, Wealth & Material Life",
  family_home: "Family, Roots & Home",
  autonomy_power: "Will, Courage & Power",
  energy_wellbeing: "Vitality, Habits & Wellbeing",
  choices_obstacles: "Challenges, Choices & Character",
  timing: "Timing, Transits & Turning Points",
};

const STRENGTH: Record<InterpretedProposition["strength"], number> = {
  dominant: 4, strong: 3, moderate: 2, subtle: 1,
};

const unique = (items: string[]): string[] => Array.from(new Set(items));
const evidence = (items: InterpretedProposition[]): string[] =>
  unique(items.flatMap(item => item.supportingEvidenceIds));

function weight(item: InterpretedProposition): number {
  return STRENGTH[item.strength] + Math.min(item.supportingEvidenceIds.length, 5) * 0.2;
}

function describe(item: InterpretedProposition): string {
  return `[${item.tradition}; ${item.frame}; ${item.strength}; ${item.valence}] ${item.themeTitle}: ${item.plainLanguageMeaning} Traditional mechanism: ${item.causalMechanism}. Evidence: ${item.supportingEvidenceIds.join(", ") || "not identified"}.`;
}

const MAX_PER_SIDE = 2;

function plainPlacementLabel(item: InterpretedProposition): string {
  // Strip module suffixes ("· Ptolemaic Aspects & Orbs") and event prefixes
  const rawSubject = item.themeTitle
    .replace(/^Event\s*·\s*/i, "")
    .split(/\s+·\s+/)[0]
    .trim();

  // Make astronomical/vedic subjects concise and natural for general readers
  if (/^SAV\s+/i.test(rawSubject)) {
    return `${rawSubject.replace(/^SAV\s+/i, "")} sign support`;
  }
  if (/^(D\d+)\s+(.+)$/i.test(rawSubject)) {
    const [, div, body] = rawSubject.match(/^(D\d+)\s+(.+)$/i)!;
    const divLabel = div === "D9" ? "Navamsha (D9)" : div === "D10" ? "Dashamsha (D10)" : div;
    return `${body} in ${divLabel}`;
  }
  return rawSubject
    .replace(/\bconjunction\b/gi, "conjunct")
    .replace(/\bsemisextile\b/gi, "semisextile")
    .replace(/–/g, " with ");
}

function summarizeInsight(item: InterpretedProposition): string {
  const cleaned = item.plainLanguageMeaning
    .replace(/\[(?:TRADITION|MODERN_SYNTHESIS[^\]]*)[^\]]*\]\s*/gi, "")
    .replace(/\b(?:Psychological dynamic|Healthy expression|Defensive risk under pressure|Psychological field)\s*:\s*/gi, "")
    .replace(/\([^)]*(?:deg|orb|virupas|rupas|evaluated|configured flow|source)[^)]*\)/gi, "")
    .trim();
  const firstSentence = cleaned.split(/(?<=[.!?])\s+/)[0] ?? cleaned;
  return firstSentence.length > 180 ? `${firstSentence.slice(0, 177).trim()}…` : firstSentence;
}

export function clusterPropositionsIntoThemes(
  propositions: InterpretedProposition[]
): {
  themes: SynthesizedTheme[];
  tensions: Array<{ polarities: string; reconciliation: string; evidenceIds: string[] }>;
} {
  const groups = new Map<HumanDomain, InterpretedProposition[]>();
  for (const proposition of propositions) {
    const group = groups.get(proposition.domain) ?? [];
    group.push(proposition);
    groups.set(proposition.domain, group);
  }

  const order: HumanDomain[] = [
    "identity", "purpose_work", "autonomy_power", "resources_wealth",
    "relationships", "choices_obstacles", "family_home", "energy_wellbeing", "timing",
  ];

  const themes: SynthesizedTheme[] = [];
  const tensions: Array<{ polarities: string; reconciliation: string; evidenceIds: string[] }> = [];
  const citedAcrossReading = new Set<string>();

  for (const domain of order) {
    const group = groups.get(domain);
    if (!group?.length) continue;

    const ordered = [...group].sort((a, b) => weight(b) - weight(a));
    const leading = ordered[0];
    const supportive = ordered.filter(p => p.valence === "supportive");
    const difficult = ordered.filter(p => p.valence === "challenging" || p.valence === "transformative");
    const traditions = unique(ordered.map(p => p.tradition));
    const allEvidence = evidence(ordered);
    const combined = ordered.map(describe);

    // Pick at most MAX_PER_SIDE fresh placements per polarity, never citing the same placement twice across the reading
    const pickFresh = (items: InterpretedProposition[]): Array<{ item: InterpretedProposition; label: string }> => {
      const picked: Array<{ item: InterpretedProposition; label: string }> = [];
      const seenHere = new Set<string>();
      for (const item of items) {
        const label = plainPlacementLabel(item);
        const key = label.toLowerCase();
        if (seenHere.has(key) || citedAcrossReading.has(key)) continue;
        seenHere.add(key);
        citedAcrossReading.add(key);
        picked.push({ item, label });
        if (picked.length >= MAX_PER_SIDE) break;
      }
      return picked;
    };

    const fav = pickFresh(supportive);
    const chal = pickFresh(difficult);

    // Distinct techniques are presented as separate witnesses, not assumed
    // to independently validate one another simply by sharing a domain.
    const corroborationLevel: SynthesizedTheme["convergenceAssessment"]["corroborationLevel"] =
      traditions.length >= 2 ? "strong-corroboration"
      : ordered.length >= 2 ? "moderate-reinforcement" : "singular";

    // Group shared evidence once instead of repeating every pair of witnesses.
    const witnesses = new Map<string, InterpretedProposition[]>();
    for (const item of ordered) for (const id of item.supportingEvidenceIds) {
      const entries = witnesses.get(id) ?? [];
      entries.push(item);
      witnesses.set(id, entries);
    }
    const relationships = Array.from(witnesses.entries())
      .filter(([, items]) => items.length > 1)
      .map(([id, items]) => `${id} links ${unique(items.map(item => plainPlacementLabel(item))).join("; ")}. These share one calculated condition and are not independent confirmations.`);

    const opposing = supportive.length > 0 && difficult.length > 0;
    const favLabelText = fav.map(f => f.label).join(" and ");
    const chalLabelText = chal.map(c => c.label).join(" and ");

    const supportLine = fav.length
      ? `What supports you: ${summarizeInsight(fav[0].item)} (${favLabelText}).`
      : "";
    const frictionLine = chal.length
      ? `Where the friction is: ${summarizeInsight(chal[0].item)} (${chalLabelText}).`
      : "";

    const reconciliation =
      fav.length && chal.length
        ? `${supportLine} ${frictionLine}`
        : supportLine || frictionLine || summarizeInsight(leading);

    if (opposing && (fav.length > 0 || chal.length > 0)) {
      const polarityHeading =
        fav.length && chal.length
          ? `${DOMAIN_TITLES[domain]} (${favLabelText} vs. ${chalLabelText})`
          : `${DOMAIN_TITLES[domain]} (${favLabelText || chalLabelText})`;
      tensions.push({
        polarities: polarityHeading,
        reconciliation,
        evidenceIds: evidence([...fav.map(f => f.item), ...chal.map(c => c.item)]),
      });
    }

    const consequenceParts = [
      ...supportive.map(p => `Available strength — ${plainPlacementLabel(p)}: ${summarizeInsight(p)} Evidence: ${p.supportingEvidenceIds.join(", ")}`),
      ...difficult.map(p => `Pressure to manage — ${plainPlacementLabel(p)}: ${summarizeInsight(p)} Evidence: ${p.supportingEvidenceIds.join(", ")}`),
    ];
    if (!consequenceParts.length) consequenceParts.push(...ordered.map(p => `${p.plainLanguageMeaning} Evidence: ${p.supportingEvidenceIds.join(", ")}`));

    themes.push({
      id: `theme-${domain}`,
      humanTheme: DOMAIN_TITLES[domain],
      domain,
      primaryConclusion: `${plainPlacementLabel(leading)}: ${summarizeInsight(leading)}`,
      whyHow: [
        `Primary mechanism: ${leading.causalMechanism}.`,
        ...combined,
        ...relationships.map(r => `CROSS-METHOD CONNECTION: ${r}`),
      ].join("\n"),
      concreteManifestations: ordered.slice(0, 4).map(p =>
        `${plainPlacementLabel(p)}: ${p.plainLanguageMeaning} Evidence: ${p.supportingEvidenceIds.join(", ")}`
      ),
      consequences: consequenceParts.join("\n"),
      deeperMeaning: `Symbolic interpretation of ${DOMAIN_TITLES[domain]}: ${reconciliation}`,
      convergenceAssessment: {
        independentTraditions: traditions,
        corroborationLevel,
        tensions: opposing && reconciliation ? [reconciliation] : [],
        reconciliation,
      },
      linkedPropositionIds: ordered.map(p => p.id),
      evidenceIds: allEvidence,
    });
  }

  return { themes, tensions };
}

const slug = (value: string) => value.toLowerCase().replace(/\s+/g, "-");

function chartEvidenceIdsForItem(
  item: EvidenceItem,
  snapshot: UnifiedMasterChartSnapshot,
): string[] {
  const evidence = new Set<string>();
  const concepts = new Set(item.concepts.map(concept => concept.toLowerCase()));

  for (const id of snapshot.methodExecutionLedger.flatMap(layer => layer.evidenceIds)) {
    if (concepts.has(id.toLowerCase())) evidence.add(id);
  }

  const bodyNames = new Set<string>();
  for (const body of [item.subject, ...item.concepts]) {
    const match = body.match(/(?:^|[-\s])(sun|moon|mercury|venus|mars|jupiter|saturn|uranus|neptune|pluto|rahu|ketu|north[- ]node|south[- ]node|ascendant|descendant|midheaven)(?:$|[-\s])/i);
    if (match) bodyNames.add(slug(match[1]));
  }
  for (const body of Array.from(bodyNames)) {
    const id = `planet-${body}`;
    if (snapshot.methodExecutionLedger.some(layer => layer.evidenceIds.includes(id))) evidence.add(id);
  }

  const houses = new Set<string>();
  for (const value of [item.subject, ...item.concepts]) {
    const match = value.match(/(?:house[- ]|House\s+)(\d{1,2})/i);
    if (match) houses.add(`house-${match[1]}`);
  }
  for (const house of Array.from(houses)) {
    if (snapshot.methodExecutionLedger.some(layer => layer.evidenceIds.includes(house))) evidence.add(house);
  }

  // Some Canon outputs are calculated by a named method rather than a single
  // planet/house evidence ID. Preserve that calculation lineage as well.
  const techniqueHints = [item.technique, ...item.concepts].map(slug);
  for (const layer of snapshot.methodExecutionLedger) {
    if (techniqueHints.some(hint => hint && (layer.id.includes(hint) || layer.technique.includes(hint)))) {
      layer.evidenceIds.forEach(id => evidence.add(id));
    }
  }

  return Array.from(evidence);
}

function truncateSentence(text: string, maxChars: number): string {
  const clean = text.replace(/\s+/g, " ").trim();
  if (clean.length <= maxChars) return clean;
  const slice = clean.slice(0, maxChars - 1);
  const lastStop = Math.max(slice.lastIndexOf(". "), slice.lastIndexOf("; "), slice.lastIndexOf(" "));
  return (lastStop > maxChars * 0.55 ? slice.slice(0, lastStop) : slice).trim() + "…";
}

function strengthRank(strength: string | number | undefined): number {
  if (typeof strength === "number") return strength;
  switch (strength) {
    case "dominant": return 4;
    case "strong": return 3;

    case "subtle": return 1;

  opts: Required<Omit<CompactSynthesisOptions, "preserveFullMethodRegistry">>
): T[] {
  const baseId = methodId.replace(/^transit:/, "");
  let limit = opts.maxOtherRows;
  if (baseId === "vedic-vargas-divisional") limit = opts.maxVargaRows;
  else if (baseId === "transit-activation-layer") limit = opts.maxTransitRows;
  else if (baseId === "western-houses-and-angles") limit = opts.maxHouseRows;

  if (findings.length <= limit) return findings;

  // Keep highest-strength and core luminary/angle subjects first.
  const priorityRe = /\b(Sun|Moon|Ascendant|Descendant|Midheaven|North Node|South Node|Mercury|Venus|Mars|Jupiter|Saturn|D9|D10|Lagna|Shadbala|Bhava Bala|Mansion|Manzil|Nakshatra|Decan)\b/i;
  const sorted = [...findings].sort((a, b) => {
    const aPri = priorityRe.test(a.subject) ? 1 : 0;
    const bPri = priorityRe.test(b.subject) ? 1 : 0;
    if (bPri !== aPri) return bPri - aPri;
    return strengthRank(b.strength) - strengthRank(a.strength);
  });
  return sorted.slice(0, limit);
}

/**
 * Compact LLM packet preserving all method IDs, evidence IDs, Canon rules,
 * and sufficient finding depth across every calculated system.
 */
export function packCompactSynthesisForLLM(
  synthesis: SynthesisResult,
  options: CompactSynthesisOptions = {}
) {
  const opts = {
    maxJudgmentChars: options.maxJudgmentChars ?? 300,
    maxVargaRows: options.maxVargaRows ?? 8,
    maxTransitRows: options.maxTransitRows ?? 16,
    maxHouseRows: options.maxHouseRows ?? 12,
    maxOtherRows: options.maxOtherRows ?? 12,
  };
  const preserveFullMethodRegistry = options.preserveFullMethodRegistry ?? true;

  const allSources = Array.from(
    new Map(
      (synthesis.canonApplications ?? []).map(rule => [
        `${rule.ruleId}:${rule.sourceTitle ?? ""}`,
        {
          ruleId: rule.ruleId,
          ...(rule.sourceId ? { sourceId: rule.sourceId } : {}),
          ...(rule.sourceTitle ? { sourceTitle: rule.sourceTitle } : {}),
          ...(rule.confidence ? { confidence: rule.confidence } : {}),
        },
      ])
    ).values()
  );
  const fullSourceIndex = new Map(
    allSources.map((rule, index) => [`${rule.ruleId}:${rule.sourceTitle ?? ""}`, index])
  );

  const usedSourceKeys = new Set<string>();
  const rawOutcomes = (synthesis.methodOutcomes ?? []).map(outcome => {
    const selected = selectRepresentativeFindings(
      outcome.methodId,
      outcome.findingJudgments ?? [],
      opts
    );
    for (const f of selected) {
      for (const rule of f.rules ?? []) {
        usedSourceKeys.add(`${rule.ruleId}:${rule.sourceTitle ?? ""}`);
      }
    }
    return { outcome, selected };
  });

  const sources = preserveFullMethodRegistry
    ? allSources
    : allSources.filter(s => usedSourceKeys.has(`${s.ruleId}:${s.sourceTitle ?? ""}`));
  const activeSourceIndex = preserveFullMethodRegistry
    ? fullSourceIndex
    : new Map(sources.map((rule, index) => [`${rule.ruleId}:${rule.sourceTitle ?? ""}`, index]));

  const findingsTotal = rawOutcomes.reduce((n, item) => n + item.outcome.findingJudgments.length, 0);
  const findingsIncluded = rawOutcomes.reduce((n, item) => n + item.selected.length, 0);
  const omittedFindings = rawOutcomes.reduce(
    (n, item) => n + item.outcome.findingJudgments.length - item.selected.length,
    0
  );
  const shortenedJudgments = rawOutcomes.reduce(
    (n, item) =>
      n +
      item.selected.filter(
        f => f.judgment.replace(/\s+/g, " ").trim().length > opts.maxJudgmentChars
      ).length,
    0
  );
  const truncated =
    rawOutcomes.some(
      item =>
        item.selected.length < item.outcome.findingJudgments.length ||
        item.selected.some(
          f =>
            f.judgment.replace(/\s+/g, " ").trim().length > opts.maxJudgmentChars ||
            f.evidenceIds.length > 5
        )
    ) ||
    synthesis.methodFindings.some(
      method => preserveFullMethodRegistry && method.evidenceIds.length > 24
    ) ||
    synthesis.contradictionsAndTensions.length > 8 ||
    synthesis.themes.length > 12 ||
    synthesis.themes
      .slice(0, 12)
      .some(theme => theme.humanTheme.replace(/\s+/g, " ").trim().length > 240);

  return {
    snapshotId: synthesis.snapshotId,
    coverage: {
      summaryOnly: truncated,
      findingsTotal,
      findingsIncluded,
      omittedFindings,
      shortenedJudgments,
      truncated,
      upstreamAuditPassed: synthesis.qualityAudit?.passed ?? null,
    },
    methodFindings: (synthesis.methodFindings ?? []).map(({ findings: _raw, ...metadata }) => {
      if (preserveFullMethodRegistry) {
        return {
          ...metadata,
          evidenceIds: (metadata.evidenceIds ?? []).slice(0, 24),
        };
      }
      return {
        id: metadata.id,
        status: metadata.status,
        frame: metadata.frame,
      };
    }),
    judgmentColumns: [
      "subject",
      "judgment",
      "evidenceIds",
      "sourceRefs",
      "valence",
      "strength",
      "frame",
    ],
    methodOutcomes: rawOutcomes.map(({ outcome, selected }) => ({
      methodId: outcome.methodId,
      methodName: outcome.methodName,
      chartScope: outcome.chartScope,
      frame: outcome.frame,
      status: outcome.status,
      findingCountTotal: outcome.findingJudgments?.length ?? 0,
      findingJudgments: selected.map(f => [
        f.subject,
        truncateSentence(f.judgment ?? "", opts.maxJudgmentChars),
        (f.evidenceIds ?? []).slice(0, 5),
        Array.from(
          new Set(
            (f.rules ?? []).map(rule =>
              activeSourceIndex.get(`${rule.ruleId}:${rule.sourceTitle ?? ""}`)
            )
          )
        ).filter((idx): idx is number => typeof idx === "number"),
        f.valence,
        f.strength,
        f.frame ?? outcome.frame,
      ]),
    })),
    canonApplications: sources,
    contradictionsAndTensions: (synthesis.contradictionsAndTensions ?? []).slice(0, 8),
    themes: (synthesis.themes ?? []).slice(0, 12).map(theme => ({
      id: theme.id,
      humanTheme: truncateSentence(theme.humanTheme, 240),
    })),
    ...(synthesis.timingOverview ? { timingOverview: synthesis.timingOverview } : {}),
  };
}

export function formatCompactSynthesisContext(
  synthesis: SynthesisResult,
  options?: CompactSynthesisOptions
) {
  return [
    "UNIFIED MASTER SYNTHESIS — COMPACT METHOD FINDINGS AND CANON APPLICATIONS",
    "Rows use judgmentColumns ([subject, judgment, evidenceIds, sourceRefs, valence, strength, frame]). sourceRefs index canonApplications. Preserve method execution statuses, separate natal (unprefixed) from event (transit:), and separate God House sectors from Agent local horizon houses. Cite original evidence ID strings, never sourceRef integers.",
    JSON.stringify(packCompactSynthesisForLLM(synthesis, options)),
  ].join("\n");
}

/** Intern repeated values without dropping or shortening any evidence. */
export function encodeEvidence(value: unknown) {
  const counts = new Map<string, number>();
  const partsOf = (text: string) => text.split(/(?<=[.!?\n])(?=\s)/);
  const jsonValue = (text: string): object | undefined => {
    if (text.length < 100 || !/^[\[{]/.test(text)) return undefined;
    try {
      const parsed = JSON.parse(text);
      return parsed && typeof parsed === "object" && JSON.stringify(parsed) === text ? parsed : undefined;
    } catch { return undefined; }
  };
  const visit = (node: unknown): void => {
    if (node === null || typeof node !== "object" && typeof node !== "string") return;
    const key = JSON.stringify(node);
    if (key.length >= 100) counts.set(key, (counts.get(key) ?? 0) + 1);
    if (typeof node === "string") {
      const parsed = jsonValue(node);
      if (parsed) { visit(parsed); return; }
      const parts = partsOf(node);
      if (parts.length > 1) for (const part of parts) {
        const partKey = JSON.stringify(part);
        if (partKey.length >= 100) counts.set(partKey, (counts.get(partKey) ?? 0) + 1);
      }
    }
    if (Array.isArray(node)) node.forEach(visit);
    else if (typeof node === "object") Object.values(node).forEach(visit);
  };
  // Normalize undefined and other JSON-only details exactly once.
  const source: unknown = JSON.parse(JSON.stringify(value));
  visit(source);
  const references: unknown[] = [];
  const indices = new Map<string, number>();
  const encode = (node: unknown, inline = false, fragment = false): unknown => {
    if (node === null || typeof node !== "object" && typeof node !== "string") return node;
    const key = JSON.stringify(node);
    if (!inline && (counts.get(key) ?? 0) > 1) {

        references[index] = encode(node, true, fragment);

      return { $evidenceRef: index };

      const parsed = jsonValue(node);
      if (parsed) return { $evidenceJson: encode(parsed) };

  };

    evidenceFormat: "Lossless shared references. Replace every object containing only $evidenceRef with references[index], recursively. export function decodeEvidence(packet: ReturnType<typeof encodeEvidence>): unknown {
  const decode = (node: unknown): unknown => {
    if (Array.isArray(node)) return node.map(decode);
    if (node && typeof node === "object") {
      const object = node as Record<string, unknown>;
      if (Object.keys(object).length === 1 && typeof object.$evidenceRef === "number")
        return decode(packet.references[object.$evidenceRef]);
      if (Object.keys(object).length === 1 && Array.isArray(object.$evidenceText))
        return object.$evidenceText.map(decode).join("");
      if (Object.keys(object).length === 1 && "$evidenceJson" in object)
        return JSON.stringify(decode(object.$evidenceJson));
      return Object.fromEntries(Object.entries(object).map(([k, v]) => [k, decode(v)]));
    }
    return node;
  };
  return decode(packet.data);
}

export function stringifyEvidence(value: unknown): string {

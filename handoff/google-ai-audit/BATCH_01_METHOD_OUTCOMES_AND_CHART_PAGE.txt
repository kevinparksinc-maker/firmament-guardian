import type {

const frameOf = (layer: LayerRecord): MethodOutcome["frame"] =>
  ["god", "agent", "transit"].includes(layer.frame)
    ? (layer.frame as MethodOutcome["frame"])
    : "shared";
function findings(layer: LayerRecord): MethodOutcomeFinding[] {
  const values = Array.isArray(layer.results)
    ? layer.results
    : layer.results == null
      ? []
      : [layer.results];
  return values.map((value, index) => ({
    evidenceId: layer.evidenceIds[index] ?? layer.evidenceIds[0] ?? layer.id,
    label: `${layer.name} finding ${index + 1}`,
    calculatedValue: typeof value === "string" ? value : JSON.stringify(value),
  }));
}
function outcome(
  layer: LayerRecord,
  snapshot: UnifiedMasterChartSnapshot,
  canon: CanonRuleApplication[]
): MethodOutcome {
  const event =
    layer.id.startsWith("transit:") ||
    layer.id === "transit-activation-layer" ||
    snapshot.inputs.chartRole === "event";
  const base = {
    methodId: layer.id,
    methodName: layer.name,
    tradition: layer.tradition,
    chartScope: event ? ("event" as const) : ("natal" as const),
    frame: frameOf(layer),
    calculatedFindings: findings(layer),
    findingJudgments: [],
    canonApplications: [],
    evidenceIds: layer.evidenceIds,
    strengthAndLimitations: layer.notes ? [layer.notes] : [],
    missingCanon: false,
  };
  if (layer.status === "reference-only")
    return {
      ...base,
      status: "reference-only",
      interpretation:
        layer.notes ??
        "Reference catalog; no executable personal placement method.",
      judgment:
        "No personal method judgment is issued from this reference-only catalog.",
      contribution: "Historical reference; excluded from interpreted themes.",
    };
  if (layer.status === "not-applicable")
    return {
      ...base,
      status: "not-applicable",
      interpretation:
        layer.notes ?? "This method's required chart scope is absent.",
      judgment: "No judgment for this scope.",
      contribution: "Not applicable; excluded from interpreted themes.",
    };
  if (layer.status === "computed-no-result")
    return {
      ...base,
      status: "no-result",
      interpretation:
        "The calculation completed and found no qualifying result.",
      judgment:
        "No positive activation was found under the method's configured conditions.",
      contribution:
        "Successful empty calculation; no personal indication inferred.",
    };
  if (layer.status !== "computed-with-result")
    return {
      ...base,
      status: "failed",
      interpretation:
        layer.notes ?? layer.errorCode ?? "The calculation is unavailable.",
      judgment: "Judgment blocked by the calculation failure.",
      contribution: "Calculation status is retained in the complete raw snapshot for AI interpretation.",
    };
  try {
    const judgments = applyMethodRules(layer, snapshot);
    if (!judgments.length)
      throw new Error(
        "METHOD_RULE_MISSING: No applicable finding judgments were produced."
      );
    const allIds = new Set(layer.evidenceIds);
    const linked = canon.filter(rule =>
      rule.calculationEvidenceIds.some(id => allIds.has(id))
    );
    // Link retrieved chart-specific doctrine to the outcome's actual judgments.
    // Detached rules remain unlinked and fail the final audit.
    for (const judgment of judgments)
      judgment.rules.push(
        ...linked.filter(rule =>
          rule.calculationEvidenceIds.some(id =>
            judgment.evidenceIds.includes(id)
          )
        )
      );
    // These producers already calculate relational/condition testimony from
    // explicit chart relationships. Admit that testimony as a judged modifier,
    // once in its owning method, rather than re-normalizing the raw snapshot.
    const owner: Record<string, string> = {
      "firmament-traditional-mechanisms": "western-planets-and-signs",
      "firmament-relational-chain": "western-houses-and-angles",
      "firmament-vedic-yogas": "genesis-behavioral-patterns",
      "firmament-arabic-persian-tradition": "arabic-hermetic-lots",
    };
    const localId = layer.id.replace(/^transit:/, "");
    const scopeIds = new Set(
      snapshot.methodExecutionLedger
        .filter(
          m => m.id.startsWith("transit:") === layer.id.startsWith("transit:")
        )
        .flatMap(m => m.evidenceIds)
    );
    for (const rule of canon.filter(
      rule =>
        owner[rule.ruleId] === localId &&
        rule.calculationEvidenceIds.some(id => scopeIds.has(id)) &&
        rule.implementationStatus !== "incomplete" &&
        rule.confidence !== "reference-only"
    )) {
      const ids = rule.calculationEvidenceIds.filter(id => scopeIds.has(id));
      const supportive = [
        "mutual-reception",
        "unilateral-reception",
        "translation-of-light",
        "collection-of-light",
        "cazimi",
      ].includes(rule.technique);
      const challenging = [
        "combustion",
        "besiegement",
        "prohibition",
        "frustration",
        "void-of-course",
      ].includes(rule.technique);
      const techniqueLabels: Record<string, string> = {
        "mutual-reception": "two placements actively supporting each other",
        "unilateral-reception": "one placement hosting and backing another",
        "translation-of-light": "a faster mover carrying momentum between two placements",
        "collection-of-light": "a shared planet bringing two separate placements into agreement",
        "cazimi": "close solar alignment sharpening focus",
        "combustion": "solar intensity overshadowing and internalizing expression",
        "besiegement": "pressure closing in from both sides",
        "prohibition": "an intervening influence interrupting direct follow-through",
        "frustration": "effort shifting direction before completion",
        "void-of-course": "a quiet pause before the next decisive push",
      };
      judgments.push({
        id: `prop-method-${layer.id}-canon-${rule.canonEvidenceId}`,
        subject: rule.subject,
        domain: judgments[0].domain,
        judgment: rule.statement,
        mechanism: techniqueLabels[rule.technique] ?? rule.technique,
        valence: supportive
          ? "supportive"
          : challenging
            ? "challenging"
            : "neutral",
        strength: "moderate",
        evidenceIds: ids,
        rules: [rule],
      });
    }
    const applications = Array.from(
      new Map(
        judgments
          .flatMap(j => j.rules)
          .map(rule => [rule.canonEvidenceId, rule])
      ).values()
    );
    const missingIds = layer.evidenceIds.filter(
      id => !judgments.some(j => j.evidenceIds.includes(id))
    );
    if (missingIds.length)
      throw new Error(
        `METHOD_EVIDENCE_UNINTERPRETED: ${missingIds.join(", ")}`
      );
    return {
      ...base,
      status: "interpreted",
      findingJudgments: judgments,
      canonApplications: applications,
      interpretation: judgments.map(j => j.judgment).join("\n\n"),
      judgment: judgments.map(j => j.judgment).join("\n\n"),
      contribution: `${judgments.length} finding-specific judgments contribute to ${Array.from(new Set(judgments.map(j => j.domain))).join(", ")}.`,
      strengthAndLimitations: [
        ...base.strengthAndLimitations,
        ...judgments.flatMap(j => (j.limitation ? [j.limitation] : [])),
      ],
    };
  } catch (error) {
    const reason = error instanceof Error ? error.message : String(error);
    return {
      ...base,
      status: "failed",
      missingCanon: reason.includes("RULE_MISSING") || reason.includes("CANON"),
      interpretation: reason,
      judgment:
        "No traditional judgment issued: this method failed rule application.",
      contribution:
        "Rule interpretation failed; calculated values remain available in the raw snapshot for AI interpretation.",
      strengthAndLimitations: [...base.strengthAndLimitations, reason],
    };
  }
}
export function buildMethodOutcomes(
  snapshot: UnifiedMasterChartSnapshot,
  canonApplications: CanonRuleApplication[]
): MethodOutcome[] {
  return snapshot.methodExecutionLedger.map(layer =>
    outcome(layer, snapshot, canonApplications)
  );
}

export function buildPlacementPackets(

    const bodyMatch = new RegExp(`(^|[^A-Za-z])${escaped}([^A-Za-z]|$)`, "i");
    const slug = row.name.toLowerCase().replace(/\s+/g, "-");

        value: {
          display: row.display,
          longitude: row.longitude,
          house: row.house,
          godHouse: row.godHouse,

          retrograde: row.retrograde,

      .filter(
        o => o.status === "interpreted" && !o.methodId.startsWith("transit:")
      )
      .flatMap(outcome =>
        outcome.findingJudgments
          .filter(
            f =>
              (!outcome.methodId.includes("dasha-and-ashtakavarga") ||
                f.subject.startsWith("SAV ")) &&
              (bodyMatch.test(f.subject) ||
                f.evidenceIds.some(
                  id => id.includes(`-${slug}-`) || id.endsWith(`-${slug}`)
                ) ||
                f.subject === `House ${row.agentHouse ?? row.house}` ||
                f.subject === `SAV ${row.display.split(" ")[0]}`)
          )
          .map(f => ({
            findingId: f.id,
            methodId: outcome.methodId,
            frame: f.frame ?? outcome.frame,
            judgment: f.judgment,
            mechanism: f.mechanism,
            evidenceIds: f.evidenceIds,
            sources: Array.from(new Set(f.rules.map(r => r.sourceTitle))),
            limitation: f.limitation,
          }))
      );
    if (!judgments.length)
      throw new Error(`PLACEMENT_JUDGMENTS_MISSING: ${row.name}`);
    for (const j of judgments)
      for (const id of j.evidenceIds) {
        if (!facts.some(f => f.id === id && f.layerId === j.methodId))
          facts.push({
            id,
            layerId: j.methodId,
            value: {
              judgment: j.judgment,
              mechanism: j.mechanism,

              sources: j.sources,
              limitation: j.limitation,

      },
    };
  });
}

/** Retain every relevant judgment once; repeated doctrine is interned losslessly. */
export function packPlacementMiniPrompt(
  packets: ReturnType<typeof buildPlacementPackets>
) {
  const unique = Array.from(
    new Map(
      packets.flatMap(p => p.judgments).map(j => [j.findingId, j])
    ).values()
  );
  const split = (text: string) => text.split(/(?<=[.!?])(?=\s)/);
  const counts = new Map<string, number>();
  for (const j of unique)
    for (const part of split(j.judgment))
      counts.set(part, (counts.get(part) ?? 0) + 1);
  const maxReferenceLength = String(counts.size).length;
  const textTable = Array.from(counts)
    .filter(([part, count]) => {
      const literalLength = JSON.stringify(part).length;
      return (count - 1) * literalLength > count * maxReferenceLength + 1;
    })
    .map(([part]) => part);
  const textIndex = new Map(textTable.map((part, i) => [part, i]));
  const sources = Array.from(new Set(unique.flatMap(j => j.sources)));
  const findingIndex = new Map(unique.map((j, i) => [j.findingId, i]));
  return {
    textTable,
    sources,
    findingColumns: [
      "judgmentParts",
      "methodId",
      "evidenceIds",
      "frame",
      "sourceRefs",
      "limitation",
    ],
    findings: unique.map(j => [
      split(j.judgment).map(part => textIndex.get(part) ?? part),
      j.methodId,
      j.evidenceIds,
      j.frame,
      j.sources.map(s => sources.indexOf(s)),
      j.limitation ?? null,
    ]),
    placements: packets.map(p => ({
      name: p.name,
      position: p.facts[0].value,
      placementEvidenceId: p.placementEvidenceId,
      facts: p.facts.map(f => ({ id: f.id, layerId: f.layerId })),
      findingRefs: p.judgments.map(j => findingIndex.get(j.findingId)),
    })),
  };
}

export const MINI_REQUEST_CHAR_LIMIT = 14000;
/** One compact outcome per relevant method; full judgments remain in the evidence drawer. */
export function boundedPlacementPacket(
  packets: ReturnType<typeof buildPlacementPackets>,
  limit = 260
) {
  return packets.map(p => {
    const methods = Array.from(new Set(p.judgments.map(j => j.methodId)));
    return {
      name: p.name,
      position: p.facts[0].value,
      placementEvidenceId: p.placementEvidenceId,
      methods: methods.map(methodId => {
        const matching = p.judgments.filter(j => j.methodId === methodId);
        const selected =
          matching.find(j => j.mechanism.includes(p.name)) ?? matching[0];
        // Exact excerpts from executed rules; never a generic invented replacement.
        const statement =
          selected.judgment.length <= limit
            ? selected.judgment
            : selected.judgment.slice(0, limit) + "…";
        return {
          methodId,
          judgment: statement,
          evidenceIds: selected.evidenceIds.slice(0, 2),
          limitation: selected.limitation,
        };
      }),
    };
  });
}

export function validatePlacementSummaries(
  value: unknown,
  packets: ReturnType<typeof buildPlacementPackets>
) {
  const result = summarySchema.parse(value);
  const expected = new Map(packets.map(p => [p.name, p]));
  if (
    result.placements.length !== expected.size ||
    new Set(result.placements.map(p => p.name)).size !== expected.size
  ) {
    throw new Error(
      "PLACEMENT_SUMMARY_INCOMPLETE: A paragraph is required for each displayed planet."
    );
  }
  for (const placement of result.placements) {
    const packet = expected.get(placement.name);
    if (!packet)
      throw new Error("PLACEMENT_SUMMARY_UNGROUNDED: Unknown planet.");
    const ids = new Set(packet.facts.map(f => f.id));
    const layers = new Set(packet.facts.map(f => f.layerId));
    if (
      placement.evidenceIds.some(id => !ids.has(id)) ||
      placement.layerIds.some(id => !layers.has(id))
    ) {
      throw new Error(
        `PLACEMENT_SUMMARY_UNGROUNDED: ${placement.name} cites unavailable evidence.`
      );
    }
    if (
      placement.paragraph.trim().split(/\s+/).length > 140 ||
      /\n\s*\n/.test(placement.paragraph)
    ) {
      throw new Error(
        `PLACEMENT_SUMMARY_TOO_LONG: ${placement.name} must fit one short paragraph.`
      );
    }
    // Sign/house boilerplate alone cannot pass as a synthesized placement paragraph.
    for (const layer of [
      "vedic-nakshatras-and-padas",
      "arabic-lunar-manazil",
      "elemental-decans",
    ]) {
      if (
        layers.has(layer) &&
        (!placement.layerIds.includes(layer) ||
          !packet.facts.some(
            f => f.layerId === layer && placement.evidenceIds.includes(f.id)
          ))
      ) {
        throw new Error(
          `PLACEMENT_SUMMARY_SHALLOW: ${placement.name} omits its calculated mansion layers.`
        );
      }
    }
    if (!placement.evidenceIds.includes(packet.placementEvidenceId)) {
      throw new Error(
        `PLACEMENT_SUMMARY_UNGROUNDED: ${placement.name} lacks its own placement.`
      );
    }
  }
  return result;
}

export async function buildChartPageReading(
  chart: ChartResult,
  mode: "natal" | "transit" | "combined",
  names: string[] = ["Sun", "Moon"]
): Promise<ChartPageReading> {
  const verified = await synthesizeChartForReading(
    chart,
    mode === "transit" ? "combined" : mode,
    "Compact individual placement card summaries"
  );
  const event = verified.snapshot.transit?.framework;
  if (mode === "transit" && !event)
    throw new Error("TRANSIT_FRAMEWORK_MISSING: Refresh the event chart.");
  const { synthesis, canonicalChart } =
    mode === "transit"
      ? await synthesizeMasterSnapshot(
          event!,
          "natal",
          "Compact event placement card summaries"
        )
      : verified;
  if (!names.length || names.length > 2 || new Set(names).size !== names.length)
    throw new Error("PLACEMENT_MINI_SCOPE: Request one or two placements.");
  const allPackets = buildPlacementPackets(canonicalChart, synthesis);
  const packets = names.map(name => {
    const packet = allPackets.find(p => p.name === name);
    if (!packet)
      throw new Error(`PLACEMENT_MINI_SCOPE: Unknown placement ${name}`);
    return packet;
  });
  const key = createHash("sha256")
    .update(
      placementMiniIdentity(chart, mode === "transit" ? "transit" : "natal") +
        JSON.stringify([mode, synthesis.snapshotId, names])
    )
    .digest("hex");
  const cached = placementCache.get(key);
  if (cached && cached.expiresAt > Date.now()) return cached.promise;
  const promise = generatePlacementParagraphs(packets, synthesis);
  placementCache.set(key, { expiresAt: Date.now() + 30 * 60_000, promise });
  while (placementCache.size > 32)
    placementCache.delete(placementCache.keys().next().value!);
  try {
    return await promise;
  } catch (error) {
    if (placementCache.get(key)?.promise === promise)
      placementCache.delete(key);
    throw error;
  }
}

async function generatePlacementParagraphs(
  packets: ReturnType<typeof buildPlacementPackets>,
  synthesis: SynthesisResult
): Promise<ChartPageReading> {
  const instruction = buildAstrologyMiniInterpreterSystem(
    `Write only the requested placement readings in candid, plainspoken, intimate second-person prose ("you / your"). Read the entire calculated packet silently and weave converging layers into one grounded everyday insight, giving natural gifts the same weight as difficult patterns.
Return JSON {"placements":[{"name":"Sun","paragraph":"...","evidenceIds":["..."],"layerIds":["..."]}]}.
REQUIREMENTS:
1. Keep each "paragraph" to one flowing paragraph of 70–110 words (max 140 words). Vary opening rhythm.
2. Start with what the reader actually does in ordinary life—what they are trying to protect or get, what comes naturally that people rely on them for, and how their behavior lands with other people.
3. Keep all astrology terminology out of "paragraph": never name planets, zodiac signs, houses, degrees, aspects, lunar mansions or stations, bounds, faces, or Lots in reader-facing text. Put all technical citations strictly inside "evidenceIds" and "layerIds".
4. Never use bullet points, headings, boilerplate openings, stock transitions, generic encouragement, recap endings, or invented events/diagnoses/motives.`
  );
  const requestFor = (limit: number) => ({
    max_tokens: 2000,
    // Keep comfortably under Render reverse proxy gateway timeouts (which abort at 60s-100s).
    timeoutMs: 25_000,
    response_format: {
      type: "json_schema" as const,
      json_schema: {
        name: "placement_minis",
        strict: true,
        schema: {
          type: "object",
          properties: {
            placements: {
              type: "array",
              items: {
                type: "object",
                properties: {
                  name: { type: "string", enum: packets.map(p => p.name) },
                  paragraph: { type: "string" },
                  evidenceIds: { type: "array", items: { type: "string" } },
                  layerIds: { type: "array", items: { type: "string" } },
                },
                required: ["name", "paragraph", "evidenceIds", "layerIds"],
                additionalProperties: false,
              },
            },
          },
          required: ["placements"],
          additionalProperties: false,
        },
      },
    },
    messages: [
      { role: "system" as const, content: instruction },
      {
        role: "user" as const,
        content: JSON.stringify({
          chartScope: synthesis.methodOutcomes[0]?.chartScope,
          placements: boundedPlacementPacket(packets, limit),
        }),
      },
    ],
  });
  // Retain all method IDs/evidence while tightening individual excerpts when
  // new Canon layers increase the packet. Split only if even minimum excerpts
  // cannot fit; never silently remove a method or invent substitute judgments.
  let request = requestFor(260);
  for (const excerptLimit of [150, 120, 95, 75, 50]) {
    if (JSON.stringify(request.messages).length <= MINI_REQUEST_CHAR_LIMIT) break;
    request = requestFor(excerptLimit);
  }
  if (
    JSON.stringify(request.messages).length > MINI_REQUEST_CHAR_LIMIT &&
    packets.length > 1
  ) {
    const first = await generatePlacementParagraphs([packets[0]], synthesis);
    const second = await generatePlacementParagraphs([packets[1]], synthesis);
    return {
      snapshotId: synthesis.snapshotId,
      dominantConclusion: synthesis.dominantConclusion,
      methodOutcomes: synthesis.methodOutcomes,
      themes: synthesis.themes,
      methodLedgerSummary: synthesis.methodLedgerSummary,
      qualityAudit: synthesis.qualityAudit,
      placements: [...first.placements, ...second.placements],
    };
  }
  if (JSON.stringify(request.messages).length > MINI_REQUEST_CHAR_LIMIT)
    throw new Error(
      "PLACEMENT_MINI_BUDGET: This placement exceeds the compact request budget."
    );
  const result = await runInterpretation("PLACEMENT_SUMMARY", synthesis, request);
  const raw = JSON.parse(interpretationText(result).replace(/^\s*```(?:json)?\s*/, "").replace(/\s*```\s*$/, ""));
  if (raw && Array.isArray(raw.placements)) {
    raw.placements = raw.placements.map((p: any) => ({
      ...p,
      paragraph: typeof p?.paragraph === "string" ? scrubNonHoraryAstrologyJargon(p.paragraph) : p?.paragraph,
    }));
  }
  const reading = validatePlacementSummaries(raw, packets);
  return {
    snapshotId: synthesis.snapshotId,
    dominantConclusion: synthesis.dominantConclusion,
    methodOutcomes: synthesis.methodOutcomes,
    themes: synthesis.themes,
    methodLedgerSummary: synthesis.methodLedgerSummary,
    qualityAudit: synthesis.qualityAudit,
    placements: reading.placements.map(p => ({
      ...p,
      evidence: packets
        .find(packet => packet.name === p.name)!
        .facts.filter(f => p.evidenceIds.includes(f.id)),
    })),
  };
}
